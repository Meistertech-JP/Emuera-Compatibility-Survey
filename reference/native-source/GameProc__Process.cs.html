  265 		public void DoScript()
  266 		{
  267 			startTime = _Library.WinmmTimer.TickCount;
  268 			state.lineCount = 0;
  269 			bool systemProcRunning = true;
  270 			try
  271 			{
  272 				while (true)
  273 				{
  274 					methodStack = 0;
  275 					systemProcRunning = true;
  276 					while (state.ScriptEnd && console.IsRunning)
  277 						runSystemProc();
  278 					if (!console.IsRunning)
  279 						break;
  280 					systemProcRunning = false;
  281 					runScriptProc();
  282 				}
  283 			}
  284 			catch (Exception ec)
  285 			{
  286 				LogicalLine currentLine = state.ErrorLine;
  287 				if (currentLine != null && currentLine is NullLine)
  288 					currentLine = null;
  289 				if (systemProcRunning)
  290 					handleExceptionInSystemProc(ec, currentLine, true);
  291 				else
  292 					handleException(ec, currentLine, true);
  293 			}
  294 		}

  347 		public SingleTerm GetValue(SuperUserDefinedMethodTerm udmt)
  348 		{
  349 			methodStack++;
  350             if (methodStack > 100)
  351             {
  352                 //StackOverflowExceptionはcatchできない上に再現性がないので発生前に一定数で打ち切る。
  353                 //環境によっては100以前にStackOverflowExceptionがでるかも？
  354                 throw new CodeEE("関数の呼び出しスタックが溢れました(無限に再帰呼び出しされていませんか？)");
  355             }
  356             SingleTerm ret = null;
  357             int temp_current = state.currentMin;
  358             state.currentMin = state.functionCount;
  359             udmt.Call.updateRetAddress(state.CurrentLine);
  360             try
  361             {
  362 				state.IntoFunction(udmt.Call, udmt.Argument, exm);
  363                 //do whileの中でthrow されたエラーはここではキャッチされない。
  364 				//#functionを全て抜けてDoScriptでキャッチされる。
  365     			runScriptProc();
  366                 ret = state.MethodReturnValue;
  367 			}
  368 			finally
  369 			{
  370 				if (udmt.Call.TopLabel.hasPrivDynamicVar)
  371 					udmt.Call.TopLabel.Out();
  372                 //1756beta2+v3:こいつらはここにないとデバッグコンソールで式中関数が事故った時に大事故になる
  373                 state.currentMin = temp_current;
  374                 methodStack--;
  375             }
  376 			return ret;
  377 		}

  442 		private void handleException(Exception exc, LogicalLine current, bool playSound)
  443 		{
  444 			console.ThrowError(playSound);
  445 			ScriptPosition position = null;
  446             if ((exc is EmueraException ee) && (ee.Position != null))
  447                 position = ee.Position;
  448             else if ((current != null) && (current.Position != null))
  449                 position = current.Position;
  450             string posString = "";
  451 			if (position != null)
  452 			{
  453 				if (position.LineNo >= 0)
  454 					posString = position.Filename + "の" + position.LineNo.ToString() + "行目で";
  455 				else
  456 					posString = position.Filename + "で";
  457 					
  458 			}
  459 			if (exc is CodeEE)
  460 			{
  461                 if (position != null)
  462 				{
  463                     if (current is InstructionLine procline && procline.FunctionCode == FunctionCode.THROW)
  464                     {
  465                         console.PrintErrorButton(posString + "THROWが発生しました", position);
  466                         printRawLine(position);
  467                         console.PrintError("THROW内容：" + exc.Message);
  468                     }
  469                     else
  470                     {
  471                         console.PrintErrorButton(posString + "エラーが発生しました:" + Program.ExeName, position);
  472 						printRawLine(position);
  473 						console.PrintError("エラー内容：" + exc.Message);
  474                     }
  475                     console.PrintError("現在の関数：@" + current.ParentLabelLine.LabelName + "（" + current.ParentLabelLine.Position.Filename + "の" + current.ParentLabelLine.Position.LineNo.ToString() + "行目）");
  476                     console.PrintError("関数呼び出しスタック：");
  477                     LogicalLine parent;
  478                     int depth = 0;
  479                     while ((parent = state.GetReturnAddressSequensial(depth++)) != null)
  480                     {
  481                         if (parent.Position != null)
  482                         {
  483                             console.PrintErrorButton("↑" + parent.Position.Filename + "の" + parent.Position.LineNo.ToString() + "行目（関数@" + parent.ParentLabelLine.LabelName + "内）", parent.Position);
  484                         }
  485                     } 
  486 				}
  487 				else
  488 				{
  489 					console.PrintError(posString + "エラーが発生しました:" + Program.ExeName);
  490 					console.PrintError(exc.Message);
  491 				}
  492 			}
  493 			else if (exc is ExeEE)
  494 			{
  495 				console.PrintError(posString + "Emueraのエラーが発生しました:" + Program.ExeName);
  496 				console.PrintError(exc.Message);
  497 			}
  498 			else
  499             {
  500 				console.PrintError(posString + "予期しないエラーが発生しました:" + Program.ExeName);
  501 				console.PrintError(exc.GetType().ToString() + ":" + exc.Message);
  502 				string[] stack = exc.StackTrace.Split('\n');
  503 				for (int i = 0; i < stack.Length; i++)
  504 				{
  505 					console.PrintError(stack[i]);
  506 				}

  508 		}
