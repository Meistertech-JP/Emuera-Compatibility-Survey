  356 		public void Return(Int64 ret)
  357 		{
  358 			if (IsFunctionMethod)
  359 			{
  360 				ReturnF(null);
  361 				return;
  362 			}
  363 			//sequential = false;//いずれにしろ順列ではない。
  364 			//呼び出し元は全部スクリプト処理
  365 			//if (functionList.Count == 0)
  366 			//{
  367 			//    throw new ExeEE("実行中の関数が存在しません");
  368 			//}
  369 			CalledFunction called = functionList[functionList.Count - 1];
  370 			if (called.IsJump)
  371 			{//JUMPした場合。即座にRETURN RESULTする。
  372                 if (called.TopLabel.hasPrivDynamicVar)
  373                     called.TopLabel.Out();
  374 				functionList.Remove(called);
  375 				if (Program.DebugMode)
  376 					console.DebugRemoveTraceLog();
  377 				Return(ret);
  378 				return;
  379 			}
  380 			if (!called.IsEvent)
  381 			{
  382                 if (called.TopLabel.hasPrivDynamicVar)
  383                     called.TopLabel.Out();
  384                 currentLine = null;
  385             }
  386 			else
  387 			{
  388                 if (called.CurrentLabel.hasPrivDynamicVar)
  389                     called.CurrentLabel.Out();
  390 				//#Singleフラグ付き関数で1が返された。
  391 				//1752 非0ではなく1と等価であることを見るように修正
  392 				//1756 全てを終了ではなく#PRIや#LATERのグループごとに修正
  393                 if (called.IsOnly)
  394                     called.FinishEvent();
  395 				else if ((called.HasSingleFlag) && (ret == 1))
  396 					called.ShiftNextGroup();
  397 				else
  398                     called.ShiftNext();//次の同名関数に進む。
  399                 currentLine = called.CurrentLabel;//関数の始点(@～～)へ移動。呼ぶべき関数が無ければnull
  400                 if (called.CurrentLabel != null)
  401                 {
  402                     lineCount++;
  403                     if (called.CurrentLabel.hasPrivDynamicVar)
  404                         called.CurrentLabel.In();
  405                 }
  406             }
  407 			if (Program.DebugMode)
  408 				console.DebugRemoveTraceLog();
  409 			//関数終了
  410             if (currentLine == null)
  411             {
  412                 currentLine = called.ReturnAddress;
  413                 functionList.RemoveAt(functionList.Count - 1);
  414 				if (currentLine == null)
  415 				{
  416 					//この時点でfunctionListは空のはず
  417 					//functionList.Clear();//全て終了。stateEndProcessに処理を返す
  418 					if (begintype != BeginType.NULL)//BEGIN XXが行なわれていれば
  419 					{
  420 						Begin();
  421 					}
  422 					return;
  423 				}
  424                 lineCount++;
  425                 //ShfitNextLine();
  426                 return;
  427 			}
  428 			else if (Program.DebugMode)
  429 			{
  430 				FunctionLabelLine label = called.CurrentLabel;
  431 				console.DebugAddTraceLog("CALL :@" + label.LabelName + ":" + label.Position.ToString() + "行目");
  432 			}
  433             lineCount++;
  434             //ShfitNextLine();
  435             return;
  436 		}

  438 		public void IntoFunction(CalledFunction call, UserDefinedFunctionArgument srcArgs, ExpressionMediator exm)
  439 		{
  440 
  441 			if (call.IsEvent)
  442 			{
  443 				foreach (CalledFunction called in functionList)
  444 				{
  445 					if (called.IsEvent)
  446 						throw new CodeEE("EVENT関数の解決前にCALLEVENT命令が行われました");
  447 				}
  448 			}
  449 			if (Program.DebugMode)
  450 			{
  451 				FunctionLabelLine label = call.CurrentLabel;
  452 				if (call.IsJump)
  453 					console.DebugAddTraceLog("JUMP :@" + label.LabelName + ":" + label.Position.ToString() + "行目");
  454 				else
  455 					console.DebugAddTraceLog("CALL :@" + label.LabelName + ":" + label.Position.ToString() + "行目");
  456 			}
  457             if (srcArgs != null)
  458             {
  459                 //引数の値を確定させる
  460                 srcArgs.SetTransporter(exm);
  461                 //プライベート変数更新
  462                 if (call.TopLabel.hasPrivDynamicVar)
  463                     call.TopLabel.In();
  464                 //更新した変数へ引数を代入
  465                 for (int i = 0; i < call.TopLabel.Arg.Length; i++)
  466                 {
  467                     if (srcArgs.Arguments[i] != null)
  468                     {
  469 						if (call.TopLabel.Arg[i].Identifier.IsReference)
  470 							((ReferenceToken)(call.TopLabel.Arg[i].Identifier)).SetRef(srcArgs.TransporterRef[i]);
  471                         else if (srcArgs.Arguments[i].GetOperandType() == typeof(Int64))
  472                             call.TopLabel.Arg[i].SetValue(srcArgs.TransporterInt[i], exm);
  473                         else
  474                             call.TopLabel.Arg[i].SetValue(srcArgs.TransporterStr[i], exm);
  475                     }
  476                 }
  477             }
  478             else//こっちに来るのはシステムからの呼び出し=引数は存在しない関数のみ ifネストの外に出していい気もしないでもないがはてさて
  479             {
  480                 //プライベート変数更新
  481                 if (call.TopLabel.hasPrivDynamicVar)
  482                     call.TopLabel.In();
  483             }
  484 			functionList.Add(call);
  485 			//sequential = false;
  486             currentLine = call.CurrentLabel;
  487             lineCount++;
  488             //ShfitNextLine();
  489         }
