   69 		private bool loadHeaderFile(string filepath, string filename)
   70 		{
   71 			StringStream st;
   72 			ScriptPosition position = null;
   73 			//EraStreamReader eReader = new EraStreamReader(false);
   74 			//1815修正 _rename.csvの適用
   75 			//eramakerEXの仕様的には.ERHに適用するのはおかしいけど、もうEmueraの仕様になっちゃってるのでしかたないか
   76 			EraStreamReader eReader = new EraStreamReader(true);
   77 
   78 			if (!eReader.Open(filepath, filename))
   79 			{
   80 				throw new CodeEE(eReader.Filename + "のオープンに失敗しました");
   81 				//return false;
   82 			}
   83 			try
   84 			{
   85 				while ((st = eReader.ReadEnabledLine()) != null)
   86 				{
   87 					if (!noError)
   88 						return false;
   89 					position = new ScriptPosition(filename, eReader.LineNo);
   90 					LexicalAnalyzer.SkipWhiteSpace(st);
   91 					if (st.Current != '#')
   92 						throw new CodeEE("ヘッダーの中に#で始まらない行があります", position);
   93 					st.ShiftNext();
   94 					string sharpID = LexicalAnalyzer.ReadSingleIdentifier(st);
   95 					if (sharpID == null)
   96 					{
   97 						ParserMediator.Warn("解釈できない#行です", position, 1);
   98 						return false;
   99 					}
  100 					if (Config.ICFunction)
  101 						sharpID = sharpID.ToUpper();
  102 					LexicalAnalyzer.SkipWhiteSpace(st);
  103 					switch (sharpID)
  104 					{
  105 						case "DEFINE":
  106 							analyzeSharpDefine(st, position);
  107 							break;
  108 						case "FUNCTION":
  109 						case "FUNCTIONS":
  110 							analyzeSharpFunction(st, position, sharpID == "FUNCTIONS");
  111 							break;
  112 						case "DIM":
  113 						case "DIMS":
  114 							//1822 #DIMは保留しておいて後でまとめてやる
  115 							{
  116 								WordCollection wc = LexicalAnalyzer.Analyse(st, LexEndWith.EoL, LexAnalyzeFlag.AllowAssignment);
  117 								dimlines.Enqueue(new DimLineWC(wc, sharpID == "DIMS", false, position));
  118 							}
  119 							//analyzeSharpDim(st, position, sharpID == "DIMS");
  120 							break;
  121 						default:
  122 							throw new CodeEE("#" + sharpID + "は解釈できないプリプロセッサです", position);
  123 					}
  124 				}
  125 			}
  126 			catch (CodeEE e)
  127 			{
  128 				if (e.Position != null)
  129 					position = e.Position;
  130 				ParserMediator.Warn(e.Message, position, 2);
  131 				return false;
  132 			}
  133 			finally
  134 			{
  135 				eReader.Close();
  136 			}
  137 			return true;
  138 		}

  147 		private void analyzeSharpDefine(StringStream st, ScriptPosition position)
  148 		{
  149 			//LexicalAnalyzer.SkipWhiteSpace(st);呼び出し前に行う。
  150 			string srcID = LexicalAnalyzer.ReadSingleIdentifier(st);
  151 			if (srcID == null)
  152 				throw new CodeEE("置換元の識別子がありません", position);
  153 			if (Config.ICVariable)
  154 				srcID = srcID.ToUpper();
  155 
  156             //ここで名称重複判定しないと、大変なことになる
  157             string errMes = "";
  158             int errLevel = -1;
  159             idDic.CheckUserMacroName(ref errMes, ref errLevel, srcID);
  160             if (errLevel >= 0)
  161             {
  162                 ParserMediator.Warn(errMes, position, errLevel);
  163                 if (errLevel >= 2)
  164                 {
  165                     noError = false;
  166                     return;
  167                 }
  168             }
  169             
  170             bool hasArg = st.Current == '(';//引数を指定する場合には直後に(が続いていなければならない。ホワイトスペースも禁止。
  171 			//1808a3 代入演算子許可（関数宣言用）
  172 			WordCollection wc = LexicalAnalyzer.Analyse(st, LexEndWith.EoL, LexAnalyzeFlag.AllowAssignment);
  173 			if (wc.EOL)
  174 			{
  175 				//throw new CodeEE("置換先の式がありません", position);
  176 				//1808a3 空マクロの許可
  177 				DefineMacro nullmac = new DefineMacro(srcID, new WordCollection(), 0);
  178 				idDic.AddMacro(nullmac);
  179 				return;
  180 			}
  181 
  182 			List<string> argID = new List<string>();
  183 			if (hasArg)//関数型マクロの引数解析
  184 			{
  185 				wc.ShiftNext();//'('を読み飛ばす
  186 				if (wc.Current.Type == ')')
  187 					throw new CodeEE("関数型マクロの引数を0個にすることはできません", position);
  188 				while (!wc.EOL)
  189 				{
  190 					IdentifierWord word = wc.Current as IdentifierWord;
  191 					if (word == null)
  192 						throw new CodeEE("置換元の引数指定の書式が間違っています", position);
  193 					word.SetIsMacro();
  194 					string id = word.Code;
  195 					if (argID.Contains(id))
  196 						throw new CodeEE("置換元の引数に同じ文字が2回以上使われています", position);
  197 					argID.Add(id);
  198 					wc.ShiftNext();
  199 					if (wc.Current.Type == ',')
  200 					{
  201 						wc.ShiftNext();
  202 						continue;
  203 					}
  204 					if (wc.Current.Type == ')')
  205 						break;
  206 					throw new CodeEE("置換元の引数指定の書式が間違っています", position);
  207 				}
  208 				if (wc.EOL)
  209 					throw new CodeEE("')'が閉じられていません", position);
  210 
  211 				wc.ShiftNext();
  212 			}
  213 			if (wc.EOL)
  214 				throw new CodeEE("置換先の式がありません", position);
  215 			WordCollection destWc = new WordCollection();
  216 			while (!wc.EOL)
  217 			{
  218 				destWc.Add(wc.Current);
  219 				wc.ShiftNext();
  220 			}
  221 			if (hasArg)//関数型マクロの引数セット
  222 			{
  223 				while (!destWc.EOL)
  224 				{
  225 					IdentifierWord word = destWc.Current as IdentifierWord;
  226 					if (word == null)
  227 					{
  228 						destWc.ShiftNext();
  229 						continue;
  230 					}
  231 					for (int i = 0; i < argID.Count; i++)
  232 					{
  233 						if (string.Equals(word.Code, argID[i], Config.SCVariable))
  234 						{
  235 							destWc.Remove();
  236 							destWc.Insert(new MacroWord(i));
  237 							break;
  238 						}
  239 					}
  240 					destWc.ShiftNext();
  241 				}
  242 				destWc.Pointer = 0;
  243 			}
  244 			if (hasArg)//1808a3 関数型マクロの封印
  245 				throw new CodeEE("関数型マクロは宣言できません", position);
  246 			DefineMacro mac = new DefineMacro(srcID, destWc, argID.Count);
  247 			idDic.AddMacro(mac);
  248 		}

  264 		//1822 #DIMだけまとめておいて後で処理
  265 		private bool analyzeSharpDimLines()
  266 		{
  267 			bool noError = true;
  268 			bool tryAgain = true;
  269 			while (dimlines.Count > 0)
  270 			{
  271 				int count = dimlines.Count;
  272 				for (int i = 0; i < count; i++)
  273 				{
  274 					DimLineWC dimline = dimlines.Dequeue();
  275 					try
  276 					{
  277 						UserDefinedVariableData data = UserDefinedVariableData.Create(dimline);
  278 						if (data.Reference)
  279 							throw new NotImplCodeEE();
  280 						VariableToken var = null;
  281 						if (data.CharaData)
  282 							var = parentProcess.VEvaluator.VariableData.CreateUserDefCharaVariable(data);
  283 						else
  284 							var = parentProcess.VEvaluator.VariableData.CreateUserDefVariable(data);
  285 						idDic.AddUseDefinedVariable(var);
  286 					}
  287 					catch (IdentifierNotFoundCodeEE e)
  288 					{
  289 						//繰り返すことで解決する見込みがあるならキューの最後に追加
  290 						if (tryAgain)
  291 						{
  292 							dimline.WC.Pointer = 0;
  293 							dimlines.Enqueue(dimline);
  294 						}
  295 						else
  296 						{
  297 							ParserMediator.Warn(e.Message, dimline.SC, 2);
  298 							noError = true;
  299 						}
  300 					}
  301 					catch (CodeEE e)
  302 					{
  303 						ParserMediator.Warn(e.Message, dimline.SC, 2);
  304 						noError = false;
  305 					}
  306 				}
  307 				if (dimlines.Count == count)
  308 					tryAgain = false;
  309 			}
  310 			return noError;
  311 		}
