  509                 //    throw new CodeEE("設定によりシステム一文字数値変数の使用が禁止されています(呼び出された変数：" + ret.Name +")");
  510                 if (ret.IsForbid)
  511                 {
  512 					if(!ret.CanForbid)
  513 						throw new ExeEE("CanForbidでない変数\"" + ret.Name +"\"にIsForbidがついている");
  514                     throw new CodeEE("呼び出された変数\"" + ret.Name +"\"は設定により使用が禁止されています");
  515                 }
  516 				if (subKey != null)
  517 					throw new CodeEE("ローカル変数でない変数" + key + "に対して@が使われました");
  518                 return ret;
  519             }
  520 			if (subKey != null)
  521 				throw new CodeEE("@の使い方が不正です");
  522 			return null;
  523 		}

  562 		public IOperandTerm GetFunctionMethod(LabelDictionary labelDic, string codeStr, IOperandTerm[] arguments, bool userDefinedOnly)
  563 		{
  564 			if (Config.ICFunction)
  565 				codeStr = codeStr.ToUpper();
  566 			if (arguments == null)//引数なし、名前のみの探索
  567 			{
  568 				if (refmethodDic.ContainsKey(codeStr))
  569 					return new UserDefinedRefMethodNoArgTerm(refmethodDic[codeStr]);
  570 				return null;
  571 			}
  572 			if ((labelDic != null) && (labelDic.Initialized))
  573 			{
  574 				if (refmethodDic.ContainsKey(codeStr))
  575 					return new UserDefinedRefMethodTerm(refmethodDic[codeStr], arguments);
  576 				FunctionLabelLine func = labelDic.GetNonEventLabel(codeStr);
  577 				if (func != null)
  578 				{
  579 					if (userDefinedOnly && !func.IsMethod)
  580 					{
  581 						throw new CodeEE("#FUNCTIONが指定されていない関数\"@" + func.LabelName + "\"をCALLF系命令で呼び出そうとしました");
  582 					}
  583 					if (func.IsMethod)
  584 					{
  585 						string errMes;
  586 						IOperandTerm ret = UserDefinedMethodTerm.Create(func, arguments, out errMes);
  587 						if(ret == null)
  588 							throw new CodeEE(errMes);
  589 						return ret;
  590 					}
  591 					//1.721 #FUNCTIONが定義されていない関数は組み込み関数を上書きしない方向に。 PANCTION.ERBのRANDとか。
  592 					if (!methodDic.ContainsKey(codeStr))
  593 						throw new CodeEE("#FUNCTIONが定義されていない関数(" + func.Position.Filename + ":" + func.Position.LineNo + "行目)を式中で呼び出そうとしました");
  594 				}
  595 			}
  596 			if (userDefinedOnly)
  597 				return null;
  598             if (!methodDic.TryGetValue(codeStr, out FunctionMethod method))
  599                 return null;
  600             string errmes = method.CheckArgumentType(codeStr, arguments);
  601 			if (errmes != null)
  602 				throw new CodeEE(errmes);
  603 			return new FunctionMethodTerm(method, arguments);
  604 		}

  607 		//名前リストを元に何がやりたかったのかを推定してCodeEEを投げる
  608 		//1822 DIMリストの解決中にIdentifierNotFoundCodeEEが飛んだ場合にはやり直しの可能性がある
  609 		public void ThrowException(string str, bool isFunc)
  610 		{
  611 			string idStr = str;
  612 			if(Config.ICFunction || Config.ICVariable) //片方だけなのは互換性用オプションなのでレアケースのはず。対応しない。
  613 				idStr = idStr.ToUpper();
  614 			if (disableList.Contains(idStr))
  615 				throw new CodeEE("\"" + str + "\"は#DISABLEが宣言されています");
  616 			if (!isFunc && privateDimList.Contains(idStr))
  617 				throw new IdentifierNotFoundCodeEE("変数\"" + str + "\"はこの関数中では定義されていません");
  618 			if (nameDic.ContainsKey(idStr))
  619 			{
  620 				DefinedNameType type = nameDic[idStr];
  621 				switch (type)
  622 				{
  623 					case DefinedNameType.Reserved:
  624 						throw new CodeEE("Emueraの予約語\"" + str + "\"が不正な使われ方をしています");
  625 					case DefinedNameType.SystemVariable:
  626 					case DefinedNameType.UserGlobalVariable:
  627 						if (isFunc)
  628 							throw new CodeEE("変数名\"" + str + "\"が関数のように使われています");
  629 						break;
  630 					case DefinedNameType.SystemMethod:
  631 					case DefinedNameType.UserRefMethod:
  632 						if (!isFunc)
  633 							throw new CodeEE("関数名\"" + str + "\"が変数のように使われています");
  634 						break;
  635 					case DefinedNameType.UserMacro:
  636 						throw new CodeEE("予期しないマクロ名\"" + str + "\"です");
  637 					case DefinedNameType.SystemInstrument:
  638 						if (isFunc)
  639 							throw new CodeEE("命令名\"" + str + "\"が関数のように使われています");
  640 						else
  641 							throw new CodeEE("命令名\"" + str + "\"が変数のように使われています");
  642 			
  643 				}
  644 			}
  645 			throw new IdentifierNotFoundCodeEE("\"" + idStr + "\"は解釈できない識別子です");
  646 		}
