  243 		public void GDrawGWithMask(GraphicsImage srcGra, GraphicsImage maskGra, Point destPoint)
  244 		{
  245 			if (g == null)
  246 				throw new NullReferenceException();
  247 			Bitmap destImg = this.GetBitmap();
  248 			byte[] srcBytes = BytesFromBitmap(srcGra.GetBitmap());
  249 			byte[] srcMaskBytes = BytesFromBitmap(maskGra.GetBitmap());
  250 			//Rectangle destRect = new Rectangle(destPoint.X, destPoint.Y, srcGra.Width, srcGra.Height);
  251 
  252 			System.Drawing.Imaging.BitmapData bmpData =
  253 				destImg.LockBits(new Rectangle(0,0, destImg.Width,destImg.Height),
  254 				System.Drawing.Imaging.ImageLockMode.ReadWrite,
  255 				PixelFormat.Format32bppArgb);
  256 			try
  257 			{
  258 				IntPtr ptr = bmpData.Scan0;
  259 				byte[] pixels = new byte[bmpData.Stride * destImg.Height];
  260 				System.Runtime.InteropServices.Marshal.Copy(ptr, pixels, 0, pixels.Length);
  261 
  262 
  263 				for (int y = 0; y < srcGra.Height; y++)
  264 				{
  265 
  266 					int destIndex = ((destPoint.Y + y) * destImg.Width + destPoint.X) * 4;
  267 					int srcIndex = ((0 + y) * srcGra.Width + 0) * 4;
  268 					for (int x = 0; x < srcGra.Width; x++)
  269 					{
  270 						if (srcMaskBytes[srcIndex] == 255)//完全不透明
  271 						{
  272 							pixels[destIndex++] = srcBytes[srcIndex++];
  273 							pixels[destIndex++] = srcBytes[srcIndex++];
  274 							pixels[destIndex++] = srcBytes[srcIndex++];
  275 							pixels[destIndex++] = srcBytes[srcIndex++];
  276 						}
  277 						else if (srcMaskBytes[srcIndex] == 0)//完全透明
  278 						{
  279 							destIndex += 4;
  280 							srcIndex += 4;
  281 						}
  282 						else//半透明 alpha/255ではなく（alpha+1）/256で計算しているがたぶん誤差
  283 						{
  284 							int mask = srcMaskBytes[srcIndex]; mask++;
  285 							pixels[destIndex] = (byte)((srcBytes[srcIndex] * mask + pixels[destIndex] * (256 - mask)) >> 8); srcIndex++; destIndex++;
  286 							pixels[destIndex] = (byte)((srcBytes[srcIndex] * mask + pixels[destIndex] * (256 - mask)) >> 8); srcIndex++; destIndex++;

  289 						}
  290 					}
  291 				}

  293 				// Bitmapへコピー

  295 			}

  324 		private static byte[] BytesFromBitmap(Bitmap bmp)
  325 		{
  326 			BitmapData bmpData = bmp.LockBits(
  327 			  new Rectangle(0, 0, bmp.Width, bmp.Height),
  328 			  ImageLockMode.ReadOnly,  // 書き込むときはReadAndWriteで
  329 			  PixelFormat.Format32bppArgb
  330 			);
  331 			if (bmpData.Stride < 0)
  332 				throw new Exception();//変な形式のが送られてくることはありえないはずだが一応
  333 			byte[] pixels = new byte[bmpData.Stride * bmp.Height];
  334 			try
  335 			{ 
  336 				IntPtr ptr = bmpData.Scan0;
  337 				System.Runtime.InteropServices.Marshal.Copy(ptr, pixels, 0, pixels.Length);
  338 			}
  339 			finally
  340 			{
  341 				bmp.UnlockBits(bmpData);
  342 
  343 			}
  344 			return pixels;
  345 		}

  347 		/// <summary>
  348 		/// GTOARRAY int ID, var array
  349 		/// エラーチェックは呼び出し元でのみ行う
  350 		/// <returns></returns>
  351 		public bool GBitmapToInt64Array(Int64[,] array, int xstart, int ystart)
  352 		{
  353 			if (g == null || Bitmap == null)
  354 				throw new NullReferenceException();
  355 			int w = Bitmap.Width;
  356 			int h = Bitmap.Height;
  357 			if (xstart + w > array.GetLength(0) || ystart + h > array.GetLength(1))
  358 				return false;
  359 			Rectangle rect = new Rectangle(0, 0, w, h);
  360 			System.Drawing.Imaging.BitmapData bmpData =
  361 				Bitmap.LockBits(rect, System.Drawing.Imaging.ImageLockMode.ReadOnly,
  362 				PixelFormat.Format32bppArgb);
  363 			IntPtr ptr = bmpData.Scan0;
  364 			byte[] rgbValues = new byte[w * h * 4];
  365 			Marshal.Copy(ptr, rgbValues, 0, rgbValues.Length);
  366 			Bitmap.UnlockBits(bmpData);
  367 			int i = 0;
  368 			for (int y = 0; y < h; y++)
  369 			{
  370 				for (int x = 0; x < w; x++)
  371 				{
  372 					array[x + xstart, y + ystart] =
  373 					rgbValues[i++] + //B
  374 					(((Int64)rgbValues[i++]) << 8) + //G
  375 					(((Int64)rgbValues[i++]) << 16) + //R
  376 					(((Int64)rgbValues[i++]) << 24);  //A
  377 				}
  378 			}
  379 			return true;
  380 		}

  387 		public bool GByteArrayToBitmap(Int64[,] array, int xstart, int ystart)
  388 		{
  389 			if (g == null || Bitmap == null)
  390 				throw new NullReferenceException();
  391 			int w = Bitmap.Width;
  392 			int h = Bitmap.Height;

  394 				return false;
  395 
  396 			byte[] rgbValues = new byte[w * h * 4];
  397 			int i = 0;
  398 			for (int y = 0; y < h; y++)

  401 				{

  405 					rgbValues[i++] = (byte)((c >> 16) & 0xFF);//R

  408 			}

  410 			System.Drawing.Imaging.BitmapData bmpData =
  411 				Bitmap.LockBits(rect, System.Drawing.Imaging.ImageLockMode.WriteOnly,
  412 				PixelFormat.Format32bppArgb);

  414 			Marshal.Copy(rgbValues, 0, ptr, rgbValues.Length);
  415 			Bitmap.UnlockBits(bmpData);
  416 			return true;
