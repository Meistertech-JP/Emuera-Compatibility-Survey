  214 		/// </summary>
  215 		Int64 StartTime = -1;
  216 		uint lastFrameTime = 0;
  217 		int lastFrame = -1;
  218 		private AnimeFrame GetCurrentFrame()
  219 		{
  220 			if (totaltime <= 0)
  221 				return null;
  222 #if DEBUG
  223 			if (FrameList.Count == 0)
  224 				throw new ExeEE("totaltime > 0なのにFrameListが空");
  225 			if (lastFrame >= FrameList.Count)
  226 				throw new ExeEE("SpriteAnime:最終フレームが範囲外");
  227 #endif
  228 			//一度もフレーム取得したことがない場合は現在時間を記録して最初のフレームを返す。
  229 			if (StartTime < 0)
  230 			{
  231 				StartTime = MinorShift._Library.WinmmTimer.CurrentFrameTime;
  232 				lastFrame = 0;
  233 				return FrameList[0];
  234 			}
  235 			//時間経過なしに複数回呼ばれた場合はさっき返したフレームをもう一度返す。
  236 			if (MinorShift._Library.WinmmTimer.CurrentFrameTime == lastFrameTime && lastFrame >= 0)
  237 				return FrameList[lastFrame];
  238 			//StartTimeからの経過時間をtotaltimeで剰余計算
  239 			Int64 time = (MinorShift._Library.WinmmTimer.CurrentFrameTime - StartTime) % totaltime;
  240 			//winmmtimerは一周して0になることがあり得るのでその場合の対策。C#の剰余の結果の符号は左辺値の符号に等しい。
  241 			if (time < 0)
  242 				time += totaltime;
  243 			foreach(AnimeFrame frame in FrameList)
  244 			{
  245 				time -=frame.DelayTimeMs;
  246 				if (time <= 0)
  247 				{
  248 					lastFrame = frame.index;
  249 					return frame;
  250 				}
  251 			}
  252 			//ここまでこないはず
  253 			throw new ExeEE("SpriteAnime:時間外参照");
  254 		}
