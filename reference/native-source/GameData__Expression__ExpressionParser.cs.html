   46 			if(wc == null)

   48 			List<IOperandTerm> terms = new List<IOperandTerm>();
   49 			TermEndWith termEndWith = TermEndWith.EoL;
   50 			switch (endWith)
   51 			{
   52 				case ArgsEndWith.EoL:
   53 					termEndWith = TermEndWith.Comma;
   54 					break;
   55                 //case ArgsEndWith.RightBracket:
   56                 //    termEndWith = TermEndWith.RightBracket_Comma;
   57                 //    break;
   58 				case ArgsEndWith.RightParenthesis:
   59 					termEndWith = TermEndWith.RightParenthesis_Comma;
   60 					break;
   61 			}
   62 			TermEndWith termEndWith_Assignment = termEndWith | TermEndWith.Assignment;
   63 			while (true)
   64 			{
   65 				Word word = wc.Current;
   66 				switch (word.Type)
   67 				{
   68 					case '\0':
   69                         if (endWith == ArgsEndWith.RightBracket)
   70                             throw new CodeEE("'['に対応する']'が見つかりません");
   71 						if (endWith == ArgsEndWith.RightParenthesis)
   72 							throw new CodeEE("'('に対応する')'が見つかりません");
   73 						goto end;
   74 					case ')':
   75 						if (endWith == ArgsEndWith.RightParenthesis)
   76 						{
   77 							wc.ShiftNext();
   78 							goto end;
   79 						}
   80 						throw new CodeEE("構文解析中に予期しない')'を発見しました");
   81                     case ']':
   82                         if (endWith == ArgsEndWith.RightBracket)
   83                         {
   84                             wc.ShiftNext();
   85                             goto end;
   86                         }
   87                         throw new CodeEE("構文解析中に予期しない']'を発見しました");
   88 				}
   89 				if(!isDefine)
   90 					terms.Add(ReduceExpressionTerm(wc, termEndWith));
   91 				else
   92 				{
   93 					terms.Add(ReduceExpressionTerm(wc, termEndWith_Assignment));
   94                     if (terms[terms.Count - 1] == null)
   95                         throw new CodeEE("関数定義の引数は省略できません");
   96 					if (wc.Current is OperatorWord)
   97 					{//=がある
   98 						wc.ShiftNext();
   99 						IOperandTerm term = reduceTerm(wc, false, termEndWith, VariableCode.__NULL__);
  100 						if (term == null)
  101 							throw new CodeEE("'='の後に式がありません");
  102 						if (term.GetOperandType() != terms[terms.Count - 1].GetOperandType())
  103 							throw new CodeEE("'='の前後で型が一致しません");
  104 						terms.Add(term);
  105 					}
  106 					else
  107 					{
  108 						if (terms[terms.Count - 1].GetOperandType() == typeof(Int64))
  109 							terms.Add(new NullTerm(0));
  110 						else
  111 							terms.Add(new NullTerm(""));
  112 					}
  113 				}
  114 				if (wc.Current.Type == ',')
  115 					wc.ShiftNext();
  116 			}
  117 		end:
  118             IOperandTerm[] ret = new IOperandTerm[terms.Count];
  119 			terms.CopyTo(ret);
  120 			return ret;
  121 		}

  224 		{
  225 			wc.ShiftNext();

  227 			if (symbol != null && symbol.Type == '.')
  228 			{//名前空間
  229 				throw new NotImplCodeEE();
  230 			}
  231 			else if (symbol != null && (symbol.Type == '(' || symbol.Type == '['))
  232 			{//関数
  233 				wc.ShiftNext();

  235 					throw new CodeEE("[]を使った機能はまだ実装されていません");
  236 				//引数を処理

  239 				if (mToken == null)

  241 					if (!Program.AnalysisMode)
  242 						GlobalStatic.IdentifierDictionary.ThrowException(idStr, true);
  243 					else
  244 					{

  246 							GlobalStatic.tempDic[idStr]++;
  247 						else
  248 							GlobalStatic.tempDic.Add(idStr, 1);
  249 						return new NullTerm(0);
  250 					}
  251 				}
  252 				return mToken;
  253 			}
  254 			else

  260 						return VariableParser.ReduceVariable(id, null, null, null);

  278 		private static CaseExpression reduceCaseExpression(WordCollection wc)
  279 		{
  280 			CaseExpression ret = new CaseExpression();
  281 			IdentifierWord id = wc.Current as IdentifierWord;
  282 			if ((id != null) && (id.Code.Equals("IS", Config.SCVariable)))
  283 			{
  284 				wc.ShiftNext();
  285 				ret.CaseType = CaseExpressionType.Is;
  286 				OperatorWord opWT = wc.Current as OperatorWord;
  287 				if (opWT == null)
  288 					throw new CodeEE("ISキーワードの後に演算子がありません");
  289 
  290 				OperatorCode op = opWT.Code;
  291 				if (!OperatorManager.IsBinary(op))
  292 					throw new CodeEE("ISキーワードの後の演算子が2項演算子ではありません");
  293 				wc.ShiftNext();
  294 				ret.Operator = op;
  295 				ret.LeftTerm = reduceTerm(wc, false, TermEndWith.Comma, VariableCode.__NULL__);
  296 				if (ret.LeftTerm == null)
  297 					throw new CodeEE("ISキーワードの後に式がありません");
  298 				//Type type = ret.LeftTerm.GetOperandType();
  299 				return ret;
  300 			}
  301 			ret.LeftTerm = reduceTerm(wc, true, TermEndWith.Comma, VariableCode.__NULL__);
  302 			if (ret.LeftTerm == null)
  303 				throw new CodeEE("CASEの引数は省略できません");
  304 			id = wc.Current as IdentifierWord;
  305 			if ((id != null) && (id.Code.Equals("TO", Config.SCVariable)))
  306 			{
  307 				ret.CaseType = CaseExpressionType.To;
  308 				wc.ShiftNext();
  309 				ret.RightTerm = reduceTerm(wc, true, TermEndWith.Comma, VariableCode.__NULL__);
  310 				if (ret.RightTerm == null)
  311 					throw new CodeEE("TOキーワードの後に式がありません");
  312 				id = wc.Current as IdentifierWord;
  313 				if ((id != null) && (id.Code.Equals("TO", Config.SCVariable)))
  314 					throw new CodeEE("TOキーワードが2度使われています");
  315 				if (ret.LeftTerm.GetOperandType() != ret.RightTerm.GetOperandType())
  316 					throw new CodeEE("TOキーワードの前後の型が一致していません");
  317 				return ret;
  318 			}
  319 			ret.CaseType = CaseExpressionType.Normal;
  320 			return ret;
  321 		}

  324 		/// <summary>
  325 		/// 解析器の本体
  326 		/// </summary>
  327 		/// <param name="wc"></param>
  328 		/// <param name="allowKeywordTo">TOキーワードが見つかっても良いか</param>
  329 		/// <param name="endWith">終端記号</param>
  330 		/// <returns></returns>
  331         private static IOperandTerm reduceTerm(WordCollection wc, bool allowKeywordTo, TermEndWith endWith, VariableCode varCode)
  332         {
  333             TermStack stack = new TermStack();
  334             //int termCount = 0;
  335             int ternaryCount = 0;
  336             OperatorCode formerOp = OperatorCode.NULL;
  337 			bool varArg = varCode != VariableCode.__NULL__;
  338 			do
  339 			{
  340 				Word token = wc.Current;
  341 				switch (token.Type)
  342 				{
  343 					case '\0':
  344 						goto end;
  345 					case '"'://LiteralStringWT
  346 						stack.Add(((LiteralStringWord)token).Str);
  347 						break;
  348 					case '0'://LiteralIntegerWT
  349 						stack.Add(((LiteralIntegerWord)token).Int);
  350 						break;
  351 					case 'F'://FormattedStringWT
  352 						stack.Add(ToStrFormTerm((StrFormWord)token));
  353 						break;
  354 					case 'A'://IdentifierWT
  355 						{
  356 							string idStr = (((IdentifierWord)token).Code);
  357 							if (idStr.Equals("TO", Config.SCVariable))
  358 							{
  359 								if (allowKeywordTo)
  360 									goto end;
  361 								else
  362 									throw new CodeEE("TOキーワードはここでは使用できません");
  363 							}
  364 							else if (idStr.Equals("IS", Config.SCVariable))
  365 								throw new CodeEE("ISキーワードはここでは使用できません");
  366 							stack.Add(reduceIdentifier(wc, idStr, varCode));
  367 							continue;
  368 						}
  369 
  370 					case '='://OperatorWT
  371 						{
  372 							if (varArg)
  373 								throw new CodeEE("変数の引数の読み取り中に予期しない演算子を発見しました");
  374 							OperatorCode op = ((OperatorWord)token).Code;
  375 							if (op == OperatorCode.Assignment)
  376 							{
  377 								if ((endWith & TermEndWith.Assignment) == TermEndWith.Assignment)
  378 									goto end;
  379 								throw new CodeEE("式中で代入演算子'='が使われています(等価比較には'=='を使用してください)");
  380 							}
  381 
  382 							if (formerOp == OperatorCode.Equal || formerOp == OperatorCode.Greater || formerOp == OperatorCode.Less
  383 								|| formerOp == OperatorCode.GreaterEqual || formerOp == OperatorCode.LessEqual || formerOp == OperatorCode.NotEqual)
  384 							{
  385 								if (op == OperatorCode.Equal || op == OperatorCode.Greater || op == OperatorCode.Less
  386 								|| op == OperatorCode.GreaterEqual || op == OperatorCode.LessEqual || op == OperatorCode.NotEqual)
  387 								{
  388 									ParserMediator.Warn("（構文上の注意）比較演算子が連続しています。", GlobalStatic.Process.GetScaningLine(), 0, false, false);
  389 								}
  390 							}
  391 							stack.Add(op);
  392 							formerOp = op;
  393 							if (op == OperatorCode.Ternary_a)
  394 								ternaryCount++;
  395 							else if (op == OperatorCode.Ternary_b)
  396 							{
  397 								if (ternaryCount > 0)
  398 									ternaryCount--;
  399 								else
  400 									throw new CodeEE("対応する'?'のない'#'です");
  401 							}
  402 							break;
  403 						}
  404 					case '(':
  405 						wc.ShiftNext();
  406                         IOperandTerm inTerm = reduceTerm(wc, false, TermEndWith.RightParenthesis, VariableCode.__NULL__);
  407                         if (inTerm == null)
  408                             throw new CodeEE("かっこ\"(\"～\")\"の中に式が含まれていません");
  409 						stack.Add(inTerm);
  410 						if (wc.Current.Type != ')')
  411 							throw new CodeEE("対応する')'のない'('です");
  412 						//termCount++;
  413 						wc.ShiftNext();
  414 						continue;
  415 					case ')':
  416 						if ((endWith & TermEndWith.RightParenthesis) == TermEndWith.RightParenthesis)
  417 							goto end;
  418 						throw new CodeEE("構文解釈中に予期しない記号'" + token.Type + "'を発見しました");
  419 					case ']':
  420 						if ((endWith & TermEndWith.RightBracket) == TermEndWith.RightBracket)
  421 							goto end;
  422 						throw new CodeEE("構文解釈中に予期しない記号'" + token.Type + "'を発見しました");
  423 					case ',':
  424 						if ((endWith & TermEndWith.Comma) == TermEndWith.Comma)
  425 							goto end;
  426 						throw new CodeEE("構文解釈中に予期しない記号'" + token.Type + "'を発見しました");
  427 					case 'M':
  428 						throw new ExeEE("マクロ解決失敗");
  429 					default:
  430 						throw new CodeEE("構文解釈中に予期しない記号'" + token.Type + "'を発見しました");
  431 				}
  432 				//termCount++;
  433 				wc.ShiftNext();
  434 			} while (!varArg);
  435 		end:
  436             if (ternaryCount > 0)
  437                 throw new CodeEE("'?'と'#'の数が正しく対応していません");
  438             return stack.ReduceAll();
  439         }

  443 		/// <summary>
  444         /// 式解決用クラス
  445         /// </summary>
  446         private class TermStack
  447         {
  448             /// <summary>
  449             /// 次に来るべきものの種類。
  450             /// (前置)単項演算子か値待ちなら0、二項・三項演算子待ちなら1、値待ちなら2、++、--、!に対応する値待ちの場合は3。
  451             /// </summary>
  452             int state = 0;
  453             bool hasBefore = false;
  454             bool hasAfter = false;
  455             bool waitAfter = false;
  456             Stack<Object> stack = new Stack<Object>();
  457             public void Add(OperatorCode op)
  458             {
  459                 if (state == 2 || state == 3)
  460                     throw new CodeEE("式が異常です");
  461                 if (state == 0)
  462                 {
  463                     if (!OperatorManager.IsUnary(op))
  464                         throw new CodeEE("式が異常です");
  465                     stack.Push(op);
  466                     if (op == OperatorCode.Plus || op == OperatorCode.Minus || op == OperatorCode.BitNot)
  467                         state = 2;
  468                     else
  469                         state = 3;
  470                     return;
  471                 }
  472                 if (state == 1)
  473                 {
  474                     //後置単項演算子の場合は特殊処理へ
  475                     if (OperatorManager.IsUnaryAfter(op))
  476                     {
  477                         if (hasAfter)
  478                         {
  479                             hasAfter = false;
  480                             throw new CodeEE("後置の単項演算子が複数存在しています");
  481                         }
  482                         if (hasBefore)
  483                         {
  484                             hasBefore = false;
  485                             throw new CodeEE("インクリメント・デクリメントを前置・後置両方同時に使うことはできません");
  486                         }
  487                         stack.Push(op);
  488                         reduceUnaryAfter();
  489                         //前置単項演算子が処理を待っている場合はここで解決
  490                         if (waitAfter)
  491                             reduceUnary();
  492                         hasBefore = false;
  493                         hasAfter = true;
  494                         waitAfter = false;
  495                         return;
  496                     }
  497                     if (!OperatorManager.IsBinary(op) && !OperatorManager.IsTernary(op))
  498                         throw new CodeEE("式が異常です");
  499                     //先に未解決の前置演算子解決
  500                     if (waitAfter)
  501                         reduceUnary();
  502                     int priority = OperatorManager.GetPriority(op);
  503                     //直前の計算の優先度が同じか高いなら還元。
  504                     while (lastPriority() >= priority)
  505                     {
  506                         this.reduceLastThree();
  507                     }
  508                     stack.Push(op);
  509                     state = 0;
  510                     waitAfter = false;
  511                     hasBefore = false;
  512                     hasAfter = false;
  513                     return;
  514                 }
  515                 throw new CodeEE("式が異常です");
  516             }
  517             public void Add(Int64 i) { Add(new SingleTerm(i)); }
  518             public void Add(string s) { Add(new SingleTerm(s)); }
  519             public void Add(IOperandTerm term)
  520             {
  521                 stack.Push(term);
  522                 if (state == 1)
  523                     throw new CodeEE("式が異常です");
  524                 if (state == 2)
  525                     waitAfter = true;
  526                 if (state == 3)
  527                 {
  528                     reduceUnary();
  529                     hasBefore = true;
  530                 }
  531                 state = 1;
  532                 return;
  533             }

  536             private int lastPriority()
  537             {
  538                 if (stack.Count < 3)
  539                     return -1;
  540                 object temp = (object)stack.Pop();
  541                 OperatorCode opCode = (OperatorCode)stack.Peek();
  542                 int priority = OperatorManager.GetPriority(opCode);
  543                 stack.Push(temp);
  544                 return priority;
  545             }

  547             public IOperandTerm ReduceAll()
  548             {
  549                 if (stack.Count == 0)
  550                     return null;
  551                 if (state != 1)
  552                     throw new CodeEE("式が異常です");
  553                 //単項演算子の待ちが未解決の時はここで解決
  554                 if (waitAfter)
  555                     reduceUnary();
  556                 waitAfter = false;
  557                 hasBefore = false;
  558                 hasAfter = false;
  559                 while (stack.Count > 1)
  560                 {
  561                     reduceLastThree();
  562                 }
  563                 IOperandTerm retTerm = (IOperandTerm)stack.Pop();
  564                 return retTerm;
  565             }

  567             private void reduceUnary()
  568             {
  569                 //if (stack.Count < 2)
  570                 //    throw new ExeEE("不正な時期の呼び出し");
  571                 IOperandTerm operand = (IOperandTerm)stack.Pop();
  572                 OperatorCode op = (OperatorCode)stack.Pop();
  573                 IOperandTerm newTerm = OperatorMethodManager.ReduceUnaryTerm(op, operand);
  574                 stack.Push(newTerm);
  575             }

  588             private void reduceLastThree()
  589             {
  590                 //if (stack.Count < 2)
  591                 //    throw new ExeEE("不正な時期の呼び出し");
  592                 IOperandTerm right = (IOperandTerm)stack.Pop();//後から入れたほうが右側
  593                 OperatorCode op = (OperatorCode)stack.Pop();
  594                 IOperandTerm left = (IOperandTerm)stack.Pop();
  595                 if (OperatorManager.IsTernary(op))
  596                 {
  597                     if (stack.Count > 1)
  598                     {
  599                         reduceTernary(left, right);
  600                         return;
  601                     }
  602                     throw new CodeEE("式の数が不足しています");
  603                 }
  604                 
  605                 IOperandTerm newTerm = OperatorMethodManager.ReduceBinaryTerm(op, left, right);
  606                 stack.Push(newTerm);
  607 			}

  610             {
  611                 _ = (OperatorCode)stack.Pop();
  612 				IOperandTerm newLeft = (IOperandTerm)stack.Pop();
  613 				
  614                 IOperandTerm newTerm = OperatorMethodManager.ReduceTernaryTerm(newLeft, left, right);
  615                 stack.Push(newTerm);
  616             }
