  133 		public static Int64 ReadInt64(StringStream st, bool retZero)
  134 		{
  135 			Int64 significand;
  136 			int expBase = 0;
  137 			int exponent = 0;
  138 			int stStartPos = st.CurrentPosition;
  139 			int stEndPos;
  140 			int fromBase = 10;
  141 			if (st.Current == '0')
  142 			{
  143 				char c = st.Next;
  144 				if ((c == 'x') || (c == 'X'))
  145 				{
  146 					fromBase = 16;
  147 					st.ShiftNext();
  148 					st.ShiftNext();
  149 				}
  150 				else if ((c == 'b') || (c == 'B'))
  151 				{
  152 					fromBase = 2;
  153 					st.ShiftNext();
  154 					st.ShiftNext();
  155 				}
  156 				//8進法は互換性の問題から採用しない。
  157 				//else if (dchar.IsDigit(c))
  158 				//{
  159 				//    fromBase = 8;
  160 				//    st.ShiftNext();
  161 				//}
  162 			}
  163 			if (retZero && st.Current != '+' && st.Current != '-' && !char.IsDigit(st.Current))
  164 			{
  165 				if (fromBase != 16)
  166 					return 0;
  167 				else if (!hexadecimalDigits.Contains(st.Current))
  168 					return 0;
  169 			}
  170 			significand = readDigits(st, fromBase);
  171 			if ((st.Current == 'p') || (st.Current == 'P'))
  172 				expBase = 2;
  173 			else if ((st.Current == 'e') || (st.Current == 'E'))
  174 				expBase = 10;
  175 			if (expBase != 0)
  176 			{
  177 				st.ShiftNext();
  178 				unchecked { exponent = (int)readDigits(st, fromBase); }
  179 			}
  180 			stEndPos = st.CurrentPosition;
  181 			if ((expBase != 0) && (exponent != 0))
  182 			{
  183 
  184 				double d = significand * Math.Pow(expBase, exponent);
  185 				if ((double.IsNaN(d)) || (double.IsInfinity(d)) || (d > Int64.MaxValue) || (d < Int64.MinValue))
  186 					throw new CodeEE("\"" + st.Substring(stStartPos, stEndPos) + "\"は64ビット符号付整数の範囲を超えています");

  188 			}
  189 			return significand;

  192 		private static Int64 readDigits(StringStream st, int fromBase)
  193 		{
  194 			int start = st.CurrentPosition;
  195 			//1756 正規表現を使ってみたがほぼ変わらなかったので没
  196 			//Match m = reg.Match(st.RowString, st.CurrentPosition);
  197 			//st.Jump(m.Length);
  198 			char c = st.Current;
  199 			if ((c == '-') || (c == '+'))
  200 			{
  201 				st.ShiftNext();
  202 			}
  203 			if (fromBase == 10)
  204 			{
  205 				while (!st.EOS)
  206 				{
  207 					c = st.Current;
  208 					if (char.IsDigit(c))
  209 					{
  210 						st.ShiftNext();
  211 						continue;
  212 					}
  213 					break;
  214 				}
  215 			}
  216 			else if (fromBase == 16)
  217 			{
  218 				while (!st.EOS)
  219 				{
  220 					c = st.Current;
  221 					if (char.IsDigit(c) || hexadecimalDigits.Contains(c))
  222 					{
  223 						st.ShiftNext();
  224 						continue;
  225 					}
  226 					break;
  227 				}
  228 			}
  229 			else if (fromBase == 2)
  230 			{
  231 				while (!st.EOS)
  232 				{
  233 					c = st.Current;
  234 					if (char.IsDigit(c))
  235 					{
  236 						if ((c != '0') && (c != '1'))
  237 							throw new CodeEE("二進法表記の中で使用できない文字が使われています");
  238 						st.ShiftNext();
  239 						continue;
  240 					}
  241 					break;
  242 				}
  243 			}
  244 			string strInt = st.Substring(start, st.CurrentPosition - start);
  245 			try
  246 			{
  247 				return Convert.ToInt64(strInt, fromBase);
  248 			}
  249 			catch (FormatException)
  250 			{
  251 				throw new CodeEE("\"" + strInt + "\"は整数値に変換できません");
  252 			}
  253 			catch (OverflowException)
  254 			{
  255 				throw new CodeEE("\"" + strInt + "\"は64ビット符号付き整数の範囲を超えています");
  256 			}
  257 			catch (ArgumentOutOfRangeException)
  258 			{
  259 				if (string.IsNullOrEmpty(strInt))
  260 					throw new CodeEE("数値として認識できる文字が必要です");
  261 				throw new CodeEE("文字列\"" + strInt + "\"は数値として認識できません");
  262 			}
  263 		}

  511 						if (endWith == StrEndWith.SingleQuotation)
  512 							goto end;
  513 						break;
  514 					case ',':
  515 						if ((endWith == StrEndWith.Comma) || (endWith == StrEndWith.LeftParenthesis_Bracket_Comma_Semicolon))
  516 							goto end;
  517 						break;
  518 					case '(':
  519 					case '[':
  520 					case ';':
  521 						if (endWith == StrEndWith.LeftParenthesis_Bracket_Comma_Semicolon)
  522 							goto end;
  523 						break;
  524 					case '\\'://エスケープ処理
  525 						st.ShiftNext();//\を読み飛ばす
  526 						switch (st.Current)
  527 						{
  528 							case StringStream.EndOfString:
  529 								throw new CodeEE("エスケープ文字\\の後に文字がありません");
  530 							case '\n': break;
  531 							case 's': buffer.Append(' '); break;
  532 							case 'S': buffer.Append('　'); break;
  533 							case 't': buffer.Append('\t'); break;
  534 							case 'n': buffer.Append('\n'); break;
  535 							default: buffer.Append(st.Current); break;
  536 						}
  537 						st.ShiftNext();//\の次の文字を読み飛ばす
  538 						continue;
  539 				}
  540 				buffer.Append(st.Current);
  541 				st.ShiftNext();
  542 			}
  543 		end:
  544 			return buffer.ToString();
  545 		}

  547 		/// <summary>
  548 		/// 失敗したらCodeEE。OperatorManagerには頼らない
  549 		/// OperatorCode.Assignmentを返すことがある。
  550 		/// </summary>
  551 		/// <param name="st"></param>
  552 		/// <returns></returns>
  553 		public static OperatorCode ReadOperator(StringStream st, bool allowAssignment)
  554 		{
  555 			char cur = st.Current;
  556 			st.ShiftNext();
  557 			char next = st.Current;
  558 			switch (cur)
  559 			{
  560 				case '+':
  561 					if (next == '+')
  562 					{
  563 						st.ShiftNext();
  564 						return OperatorCode.Increment;
  565 					}
  566 					return OperatorCode.Plus;
  567 				case '-':
  568 					if (next == '-')
  569 					{
  570 						st.ShiftNext();
  571 						return OperatorCode.Decrement;
  572 					}
  573 					return OperatorCode.Minus;
  574 				case '*':
  575 					return OperatorCode.Mult;
  576 				case '/':
  577 					return OperatorCode.Div;
  578 				case '%':
  579 					return OperatorCode.Mod;
  580 				case '=':
  581 					if (next == '=')
  582 					{
  583 						st.ShiftNext();
  584 						return OperatorCode.Equal;
  585 					}
  586 					if (allowAssignment)
  587 						return OperatorCode.Assignment;
  588 					throw new CodeEE("予期しない代入演算子'='を発見しました(等価比較には'=='を使用してください)");
  589 				case '!':
  590 					if (next == '=')
  591 					{
  592 						st.ShiftNext();
  593 						return OperatorCode.NotEqual;
  594 					}
  595 					else if (next == '&')
  596 					{
  597 						st.ShiftNext();
  598 						return OperatorCode.Nand;
  599 					}
  600 					else if (next == '|')
  601 					{
  602 						st.ShiftNext();
  603 						return OperatorCode.Nor;
  604 					}
  605 					return OperatorCode.Not;
  606 				case '<':
  607 					if (next == '=')
  608 					{
  609 						st.ShiftNext();
  610 						return OperatorCode.LessEqual;
  611 					}
  612 					else if (next == '<')
  613 					{
  614 						st.ShiftNext();
  615 						return OperatorCode.LeftShift;
  616 					}
  617 					return OperatorCode.Less;
  618 				case '>':
  619 					if (next == '=')
  620 					{
  621 						st.ShiftNext();
  622 						return OperatorCode.GreaterEqual;
  623 					}
  624 					else if (next == '>')
  625 					{
  626 						st.ShiftNext();
  627 						return OperatorCode.RightShift;
  628 					}
  629 					return OperatorCode.Greater;
  630 				case '|':
  631 					if (next == '|')
  632 					{
  633 						st.ShiftNext();
  634 						return OperatorCode.Or;
  635 					}
  636 					return OperatorCode.BitOr;
  637 				case '&':
  638 					if (next == '&')
  639 					{
  640 						st.ShiftNext();
  641 						return OperatorCode.And;
  642 					}
  643 					return OperatorCode.BitAnd;
  644 				case '^':
  645 					if (next == '^')
  646 					{
  647 						st.ShiftNext();
  648 						return OperatorCode.Xor;
  649 					}
  650 					return OperatorCode.BitXor;
  651 				case '~':
  652 					return OperatorCode.BitNot;
  653 				case '?':
  654 					return OperatorCode.Ternary_a;
  655 				case '#':
  656 					return OperatorCode.Ternary_b;
  657 
  658 			}
  659 			throw new CodeEE("'" + cur + "'は演算子として認識できません");
  660 		}

  675 			{
  676 				case '+':
  677 					if (next == '+')
  678 						ret = OperatorCode.Increment;
  679 					else if (next == '=')
  680 						ret = OperatorCode.Plus;
  681 					break;
  682 				case '-':
  683 					if (next == '-')
  684 						ret = OperatorCode.Decrement;
  685 					else if (next == '=')
  686 						ret = OperatorCode.Minus;
  687 					break;
  688 				case '*':

  692 				case '/':

  694 						ret = OperatorCode.Div;
  695 					break;
  696 				case '%':
  697 					if (next == '=')

  724 					}
  725 					break;
  726 				case '>':

  729 						st.ShiftNext();
  730 						if (st.Current == '=')
  731 						{
  732 							ret = OperatorCode.RightShift;
  733 							break;
  734 						}
  735 						throw new CodeEE("'>'は代入演算子として認識できません");
  736 					}
  737 					break;
  738 				case '|':
  739 					if (next == '=')
  740 						ret = OperatorCode.BitOr;
  741 					break;
  742 				case '&':
  743 					if (next == '=')
  744 						ret = OperatorCode.BitAnd;
  745 					break;
  746 				case '^':
  747 					if (next == '=')
  748 						ret = OperatorCode.BitXor;
  749 					break;
  750 			}
  751 			if (ret == OperatorCode.NULL)
  752 				throw new CodeEE("'" + cur + "'は代入演算子として認識できません");
  753 			st.ShiftNext();
  754 			return ret;
  755 		}

  839 		/// <summary>
  840 		/// 解析できるものは関数宣言や式のみ。FORM文字列や普通の文字列を送ってはいけない
  841 		/// return時にはendWithの文字がCurrentになっているはず。終端の適切さの検証は呼び出し元が行う。
  842 		/// </summary>
  843 		/// <returns></returns>
  844 		public static WordCollection Analyse(StringStream st, LexEndWith endWith, LexAnalyzeFlag flag)
  845 		{
  846 			WordCollection ret = new WordCollection();
  847 			int nestBracketS = 0;
  848 			//int nestBracketM = 0;
  849 			int nestBracketL = 0;
  850 			while (true)
  851 			{
  852 				switch (st.Current)
  853 				{
  854 					case '\n':
  855 					case '\0':
  856 						goto end;
  857 					case ' ':
  858 					case '\t':
  859 						st.ShiftNext();
  860 						continue;
  861 					case '　':
  862 						if (!Config.SystemAllowFullSpace)
  863 							throw new CodeEE("字句解析中に予期しない全角スペースを発見しました(この警告はシステムオプション「" + Config.GetConfigName(ConfigCode.SystemAllowFullSpace) + "」により無視できます)");
  864 						st.ShiftNext();
  865 						continue;
  866 					case '0':
  867 					case '1':
  868 					case '2':
  869 					case '3':
  870 					case '4':
  871 					case '5':
  872 					case '6':
  873 					case '7':
  874 					case '8':
  875 					case '9':
  876 						ret.Add(new LiteralIntegerWord(ReadInt64(st, false)));
  877 						break;
  878 					case '>':
  879 						if(endWith == LexEndWith.GreaterThan)
  880 							goto end;
  881 						goto case '+';
  882 					case '+':
  883 					case '-':
  884 					case '*':
  885 					case '/':
  886 					case '%':
  887 					case '=':
  888 					case '!':
  889 					case '<':
  890 					case '|':
  891 					case '&':
  892 					case '^':
  893 					case '~':
  894 					case '?':
  895 					case '#':
  896 						if ((nestBracketS == 0) && (nestBracketL == 0))
  897 						{
  898 							if (endWith == LexEndWith.Operator)
  899 								goto end;//代入演算子のはずである。呼び出し元がチェックするはず
  900 							else if ((endWith == LexEndWith.Percent) && (st.Current == '%'))
  901 								goto end;
  902 							else if ((endWith == LexEndWith.Question) && (st.Current == '?'))
  903 								goto end;
  904 						}
  905 						ret.Add(new OperatorWord(ReadOperator(st, (flag & LexAnalyzeFlag.AllowAssignment) == LexAnalyzeFlag.AllowAssignment)));
  906 						break;
  907 					case ')': ret.Add(new SymbolWord(')')); nestBracketS--; st.ShiftNext(); continue;
  908 					case ']': ret.Add(new SymbolWord(']')); nestBracketL--; st.ShiftNext(); continue;
  909 					case '(': ret.Add(new SymbolWord('(')); nestBracketS++; st.ShiftNext(); continue;
  910 					case '[':
  911 						if (st.Next == '[')
  912 						{
  913 							//throw new CodeEE("字句解析中に予期しない文字'[['を発見しました");
  914 							////1808alpha006 rename処理変更
  915 							//1808beta009 ここだけ戻す
  916 							//現在の処理だとここに来た時点でrename失敗確定だが警告内容を元に戻すため
  917 							if (ParserMediator.RenameDic == null)
  918 								throw new CodeEE("字句解析中に予期しない文字\"[[\"を発見しました");
  919 							int start = st.CurrentPosition;
  920 							int find = st.Find("]]");
  921 							if (find <= 2)
  922 							{
  923 								if (find == 2)
  924 									throw new CodeEE("空の[[]]です");
  925 								else
  926 									throw new CodeEE("対応する\"]]\"のない\"[[\"です");
  927 							}
  928 							string key = st.Substring(start, find + 2);
  929 							//1810 ここまでで置換できなかったものは強制エラーにする
  930 							//行連結前に置換不能で行連結より置換することができるようになったものまで置換されていたため
  931 							throw new CodeEE("字句解析中に置換(rename)できない符号" + key + "を発見しました");
  932 							//string value = null;
  933 							//if (!ParserMediator.RenameDic.TryGetValue(key, out value))
  934 							//    throw new CodeEE("字句解析中に置換(rename)できない符号" + key + "を発見しました");
  935 							//st.Replace(start, find + 2, value);
  936 							//continue;//その場から再度解析スタート
  937 						}
  938 						ret.Add(new SymbolWord('[')); nestBracketL++; st.ShiftNext(); continue;
  939 					case ':': ret.Add(new SymbolWord(':')); st.ShiftNext(); continue;
  940 					case ',':
  941 						if ((endWith == LexEndWith.Comma) && (nestBracketS == 0))// && (nestBracketL == 0))
  942 							goto end;
  943 						ret.Add(new SymbolWord(',')); st.ShiftNext(); continue;
  944 					//case '}': ret.Add(new SymbolWT('}')); nestBracketM--; continue;
  945 					//case '{': ret.Add(new SymbolWT('{')); nestBracketM++; continue;
  946 					case '\'':
  947 						if ((flag & LexAnalyzeFlag.AllowSingleQuotationStr) == LexAnalyzeFlag.AllowSingleQuotationStr)
  948 						{
  949 							st.ShiftNext();
  950 							ret.Add(new LiteralStringWord(ReadString(st, StrEndWith.SingleQuotation)));
  951 							if (st.Current != '\'')
  952 								throw new CodeEE("\'が閉じられていません");
  953 							st.ShiftNext();
  954 							break;
  955 						}
  956 						if ((flag & LexAnalyzeFlag.AnalyzePrintV) != LexAnalyzeFlag.AnalyzePrintV)
  957 						{
  958 							//AssignmentStr用特殊処理 代入文の代入演算子を探索中で'=の場合のみ許可
  959 							if ((endWith == LexEndWith.Operator) && (nestBracketS == 0) && (nestBracketL == 0) && st.Next == '=' )
  960 								goto end;
  961 							throw new CodeEE("字句解析中に予期しない文字'" + st.Current + "'を発見しました");
  962 						}
  963 						st.ShiftNext();
  964 						ret.Add(new LiteralStringWord(ReadString(st, StrEndWith.Comma)));
  965 						if (st.Current == ',')
  966 							goto case ',';//続きがあるなら,の処理へ。それ以外は行終端のはず
  967 						goto end;
  968 					case '}':
  969 						if (endWith == LexEndWith.RightCurlyBrace)
  970 							goto end;
  971 						throw new CodeEE("字句解析中に予期しない文字'" + st.Current + "'を発見しました");
  972 					case '\"':
  973 						st.ShiftNext();
  974 						ret.Add(new LiteralStringWord(ReadString(st, StrEndWith.DoubleQuotation)));
  975 						if (st.Current != '\"')
  976 							throw new CodeEE("\"が閉じられていません");
  977 						st.ShiftNext();
  978 						break;
  979 					case '@':
  980 						if (st.Next != '\"')
  981 						{
  982 							ret.Add(new SymbolWord('@'));
  983 							st.ShiftNext();
  984 							continue;
  985 						}
  986 						st.ShiftNext();
  987 						st.ShiftNext();
  988 						ret.Add(AnalyseFormattedString(st, FormStrEndWith.DoubleQuotation, false));
  989 						if (st.Current != '\"')
  990 							throw new CodeEE("\"が閉じられていません");
  991 						st.ShiftNext();
  992 						break;
  993 					case '.':
  994 						ret.Add(new SymbolWord('.'));
  995 						st.ShiftNext();
  996 						continue;
  997 
  998 					case '\\':
  999 						if (st.Next != '@')
 1000 							throw new CodeEE("字句解析中に予期しない文字'" + st.Current + "'を発見しました");
 1001 						{
 1002 							st.Jump(2);
 1003 							ret.Add(new StrFormWord(new string[] { "", "" }, new SubWord[] { AnalyseYenAt(st) }));
 1004 						}
 1005 						break;
 1006 					case '{':
 1007 					case '$':
 1008 						throw new CodeEE("字句解析中に予期しない文字'" + st.Current + "'を発見しました");
 1009 					case ';'://1807 行中コメント
 1010 						if (st.CurrentEqualTo(";#;") && Program.DebugMode)
 1011 						{
 1012 							st.Jump(3);
 1013 							break;
 1014 						}
 1015 						else if (st.CurrentEqualTo(";!;"))
 1016 						{
 1017 							st.Jump(3);
 1018 							break;
 1019 						}
 1020 						st.Seek(0, System.IO.SeekOrigin.End);
 1021 						goto end;
 1022 					default:
 1023 						{
 1024 							ret.Add(new IdentifierWord(ReadSingleIdentifier(st)));
 1025 							break;
 1026 						}
 1027 				}
 1028 			}
 1029 		end:
 1030 			if ((nestBracketS != 0) || (nestBracketL != 0))
 1031 			{
 1032 				if (nestBracketS < 0)
 1033 					throw new CodeEE("字句解析中に対応する'('のない')'を発見しました");
 1034 				else if (nestBracketS > 0)
 1035 					throw new CodeEE("字句解析中に対応する')'のない'('を発見しました");
 1036 				if (nestBracketL < 0)
 1037 					throw new CodeEE("字句解析中に対応する'['のない']'を発見しました");
 1038 				else if (nestBracketL > 0)
 1039 					throw new CodeEE("字句解析中に対応する']'のない'['を発見しました");
 1040 			}
 1041 			if (UseMacro)
 1042 				return expandMacro(ret);
 1043 			return ret;
 1044 
 1045 		}

 1047 		private static WordCollection expandMacro(WordCollection wc)
 1048 		{
 1049 			//マクロ展開
 1050 			wc.Pointer = 0;
 1051 			int count = 0;
 1052 			while (!wc.EOL)
 1053 			{
 1054 				IdentifierWord word = wc.Current as IdentifierWord;
 1055 				if (word == null)
 1056 				{
 1057 					wc.ShiftNext();
 1058 					continue;
 1059 				}
 1060 				string idStr = word.Code;
 1061 				DefineMacro macro = GlobalStatic.IdentifierDictionary.GetMacro(idStr);
 1062 				if (macro == null)
 1063 				{
 1064 					wc.ShiftNext();
 1065 					continue;
 1066 				}
 1067 				count++;
 1068 				if (count > MAX_EXPAND_MACRO)
 1069 					throw new CodeEE("マクロの展開数が1文あたりの上限" + MAX_EXPAND_MACRO.ToString() + "を超えました(自己参照・循環参照のおそれ)");
 1070 				if (!macro.HasArguments)
 1071 				{
 1072 					wc.Remove();
 1073 					wc.InsertRange(macro.Statement);
 1074 					continue;
 1075 				}
 1076 				//関数型マクロ
 1077 				wc = expandFunctionlikeMacro(macro, wc);
 1078 			}
 1079 			wc.Pointer = 0;
 1080 			return wc;
 1081 		}

 1083 		private static WordCollection expandFunctionlikeMacro(DefineMacro macro, WordCollection wc)
 1084 		{
 1085 			int macroStart = wc.Pointer;
 1086 			wc.ShiftNext();
 1087 			SymbolWord symbol = wc.Current as SymbolWord;
 1088 			if (symbol == null || symbol.Type != '(')
 1089 				throw new CodeEE("関数形式のマクロ" + macro.Keyword + "に引数がありません");
 1090 			WordCollection macroWC = macro.Statement.Clone();
 1091 			WordCollection[] args = new WordCollection[macro.ArgCount];
 1092 			//引数部読み取りループ
 1093 			for (int i = 0; i < macro.ArgCount; i++)
 1094 			{
 1095 				int macroNestBracketS = 0;
 1096 				args[i] = new WordCollection();
 1097 				while (true)
 1098 				{
 1099 					wc.ShiftNext();
 1100 					if (wc.EOL)
 1101 						throw new CodeEE("関数形式のマクロ" + macro.Keyword + "の用法が正しくありません");
 1102 					symbol = wc.Current as SymbolWord;
 1103 					if (symbol == null)
 1104 					{
 1105 						args[i].Add(wc.Current);
 1106 						continue;
 1107 					}
 1108 					switch (symbol.Type)
 1109 					{
 1110 						case '(': macroNestBracketS++; break;
 1111 						case ')':
 1112 							if (macroNestBracketS > 0)
 1113 							{
 1114 								macroNestBracketS--;
 1115 								break;
 1116 							}
 1117 							if (i != macro.ArgCount - 1)
 1118 								throw new CodeEE("関数形式のマクロ" + macro.Keyword + "の引数の数が正しくありません");
 1119 							goto exitfor;
 1120 						case ',':
 1121 							if (macroNestBracketS == 0)
 1122 								goto exitwhile;
 1123 							break;
 1124 					}
 1125 					args[i].Add(wc.Current);
 1126 				}
 1127 			exitwhile:
 1128 				if (args[i].Collection.Count == 0)
 1129 					throw new CodeEE("関数形式のマクロ" + macro.Keyword + "の引数を省略することはできません");
 1130 				continue;
 1131 			}
 1132 		//引数部読み取りループ終端
 1133 		exitfor:
 1134 			symbol = wc.Current as SymbolWord;
 1135 			if (symbol == null || symbol.Type != ')')
 1136 				throw new CodeEE("関数形式のマクロ" + macro.Keyword + "の用法が正しくありません");
 1137 			int macroLength = wc.Pointer - macroStart + 1;
 1138 			wc.Pointer = macroStart;
 1139 			for (int j = 0; j < macroLength; j++)
 1140 				wc.Collection.RemoveAt(macroStart);
 1141 			while (!macroWC.EOL)
 1142 			{
 1143 				MacroWord w = macroWC.Current as MacroWord;
 1144 				if (w == null)
 1145 				{
 1146 					macroWC.ShiftNext();
 1147 					continue;
 1148 				}
 1149 				macroWC.Remove();
 1150 				macroWC.InsertRange(args[w.Number]);
 1151 				macroWC.Pointer += args[w.Number].Collection.Count;
 1152 			}
 1153 			wc.InsertRange(macroWC);
 1154 			wc.Pointer = macroStart;
 1155 			return wc;
 1156 		}

 1158 		/// <summary>
 1159 		/// @"などの直後からの開始
 1160 		/// return時にはendWithの文字がCurrentになっているはず。終端の適切さの検証は呼び出し元が行う。
 1161 		/// </summary>
 1162 		/// <returns></returns>
 1163 		public static StrFormWord AnalyseFormattedString(StringStream st, FormStrEndWith endWith, bool trim)
 1164 		{
 1165 			List<string> strs = new List<string>();
 1166 			List<SubWord> SWTs = new List<SubWord>();
 1167 			StringBuilder buffer = new StringBuilder(100);
 1168 			while (true)
 1169 			{
 1170 				char cur = st.Current;
 1171 				switch (cur)
 1172 				{
 1173 					case '\n':
 1174 					case '\0':
 1175 						goto end;
 1176 					case '\"':
 1177 						if (endWith == FormStrEndWith.DoubleQuotation)
 1178 							goto end;
 1179 						buffer.Append(cur);
 1180 						break;
 1181 					case '#':
 1182 						if (endWith == FormStrEndWith.Sharp)
 1183 							goto end;
 1184 						buffer.Append(cur);
 1185 						break;
 1186 					case ',':
 1187 						if ((endWith == FormStrEndWith.Comma) || (endWith == FormStrEndWith.LeftParenthesis_Bracket_Comma_Semicolon))
 1188 							goto end;
 1189 						buffer.Append(cur);
 1190 						break;
 1191 					case '(':
 1192 					case '[':
 1193 					case ';':
 1194 						if (endWith == FormStrEndWith.LeftParenthesis_Bracket_Comma_Semicolon)
 1195 							goto end;
 1196 						buffer.Append(cur);
 1197 						break;
 1198 					case '%':
 1199 						strs.Add(buffer.ToString());
 1200 						buffer.Remove(0, buffer.Length);
 1201 						st.ShiftNext();
 1202 						SWTs.Add(new PercentSubWord(Analyse(st, LexEndWith.Percent, LexAnalyzeFlag.None)));
 1203 						if (st.Current != '%')
 1204 							throw new CodeEE("\'%\'が使われましたが対応する\'%\'が見つかりません");
 1205 						break;
 1206 					case '{':
 1207 						strs.Add(buffer.ToString());
 1208 						buffer.Remove(0, buffer.Length);
 1209 						st.ShiftNext();
 1210 						SWTs.Add(new CurlyBraceSubWord(Analyse(st, LexEndWith.RightCurlyBrace, LexAnalyzeFlag.None)));
 1211 						if (st.Current != '}')
 1212 							throw new CodeEE("\'{\'が使われましたが対応する\'}\'が見つかりません");
 1213 						break;
 1214 					case '*':
 1215 					case '+':
 1216 					case '=':
 1217 					case '/':
 1218 					case '$':
 1219 						if (!Config.SystemIgnoreTripleSymbol && st.TripleSymbol())
 1220 						{
 1221 							strs.Add(buffer.ToString());
 1222 							buffer.Remove(0, buffer.Length);
 1223 							st.Jump(3);
 1224 							SWTs.Add(new TripleSymbolSubWord(cur));
 1225 							continue;
 1226 						}
 1227 						else
 1228 							buffer.Append(cur);
 1229 						break;
 1230 					case '\\'://エスケープ文字の使用
 1231 
 1232 						st.ShiftNext();
 1233 						cur = st.Current;
 1234 						switch (cur)
 1235 						{
 1236 							case '\0':
 1237 								throw new CodeEE("エスケープ文字\\の後に文字がありません");
 1238 							case '\n': break;
 1239 							case 's': buffer.Append(' '); break;
 1240 							case 'S': buffer.Append('　'); break;
 1241 							case 't': buffer.Append('\t'); break;
 1242 							case 'n': buffer.Append('\n'); break;
 1243 							case '@'://\@～～?～～#～～\@
 1244 								{
 1245 									if ((endWith == FormStrEndWith.YenAt) || (endWith == FormStrEndWith.Sharp))
 1246 										goto end;
 1247 									strs.Add(buffer.ToString());
 1248 									buffer.Remove(0, buffer.Length);
 1249 									st.ShiftNext();
 1250 									SWTs.Add(AnalyseYenAt(st));
 1251 									continue;
 1252 								}
 1253 							default:
 1254 								buffer.Append(cur);
 1255 								st.ShiftNext();
 1256 								continue;
 1257 						}
 1258 						break;
 1259 					default:
 1260 						buffer.Append(cur);
 1261 						break;
 1262 				}
 1263 				st.ShiftNext();
 1264 			}
 1265 		end:
 1266 			strs.Add(buffer.ToString());
 1267 
 1268 			string[] retStr = new string[strs.Count];
 1269 			SubWord[] retSWTs = new SubWord[SWTs.Count];
 1270 			strs.CopyTo(retStr);
 1271 			SWTs.CopyTo(retSWTs);
 1272 			if (trim && retStr.Length > 0)
 1273 			{
 1274 				retStr[0] = retStr[0].TrimStart(new char[] { ' ', '\t' });
 1275 				retStr[retStr.Length - 1] = retStr[retStr.Length - 1].TrimEnd(new char[] { ' ', '\t' });
 1276 			}
 1277 			return new StrFormWord(retStr, retSWTs);
 1278 		}

 1295 			{
 1296 				if (st.Current != '@')
 1297 					throw new CodeEE("\'\\@\',\'?\'が使われましたが対応する\'#\'が見つかりません");

 1301 			}
 1302 			st.ShiftNext();

 1304 			if (st.Current != '@')
 1305 				throw new CodeEE("\'\\@\',\'?\',\'#\'が使われましたが対応する\'\\@\'が見つかりません");

 1308 		}
