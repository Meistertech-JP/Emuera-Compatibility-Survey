  382 		private string CreateTypeCString(string str, bool alignmentRight)
  383 		{
  384 			if (printCWidth == -1)
  385 				calcPrintCWidth(stringMeasure);
  386 			int length = 0;
  387 			int width;
  388 			if (str != null)
  389 				length = Config.Encode.GetByteCount(str);
  390 			int printcLength = Config.PrintCLength;
  391 			Font font;
  392 			try
  393 			{
  394 				font = new Font(Style.Fontname, Config.Font.Size, Style.FontStyle, GraphicsUnit.Pixel);
  395 			}
  396 			catch
  397 			{
  398 				return str;
  399 			}
  400 
  401 			if ((alignmentRight) && (length < printcLength))
  402 			{
  403 				str = new string(' ', printcLength - length) + str;
  404 				width = stringMeasure.GetDisplayLength(str, font);
  405 				while (width > printCWidth)
  406 				{
  407 					if (str[0] != ' ')
  408 						break;
  409 					str = str.Remove(0, 1);
  410 					width = stringMeasure.GetDisplayLength(str, font);
  411 				}
  412 			}
  413 			else if ((!alignmentRight) && (length < printcLength + 1))
  414 			{
  415 				str += new string(' ', printcLength + 1 - length);
  416 				width = stringMeasure.GetDisplayLength(str, font);
  417 				while (width > printCWidthL)
  418 				{
  419 					if (str[str.Length - 1] != ' ')
  420 						break;
  421 					str = str.Remove(str.Length - 1, 1);
  422 					width = stringMeasure.GetDisplayLength(str, font);
  423 				}
  424 			}
  425 			return str;
  426 		}

  549 			int width = 0;

  558 				bar.Remove(bar.Length - 1, 1);
  559 				width = stringMeasure.GetDisplayLength(bar.ToString(), font);
  560 			}

  630 		public ConsoleDisplayLine[] GetDisplayLines(Int64 lineNo)
  631 		{
  632 			if (lineNo < 0 || lineNo > displayLineList.Count)
  633 				return null;
  634 			int count = 0;
  635 			List<ConsoleDisplayLine> list = new List<ConsoleDisplayLine>();
  636 			for (int i = displayLineList.Count - 1; i >= 0; i--)
  637 			{
  638 				if (count == lineNo)
  639 					list.Insert(0, displayLineList[i]);
  640 				if (displayLineList[i].IsLogicalLine)
  641 					count++;
  642 				if (count > lineNo)
  643 					break;
  644 			}
  645 			if (list.Count == 0)
  646 				return null;
  647 			ConsoleDisplayLine[] ret = new ConsoleDisplayLine[list.Count];
  648 			list.CopyTo(ret);
  649 			return ret;
  650 		}
