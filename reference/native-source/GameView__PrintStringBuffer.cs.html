  325 		private ConsoleButtonString[] createButtons(List<AConsoleDisplayPart> cssList)
  326 		{
  327 			StringBuilder buf = new StringBuilder();
  328 			for (int i = 0; i < cssList.Count; i++)
  329 			{
  330 				buf.Append(cssList[i].Str);
  331 			}
  332 			List<ButtonPrimitive> bpList = ButtonStringCreator.SplitButton(buf.ToString());
  333 			ConsoleButtonString[] ret = new ConsoleButtonString[bpList.Count];
  334 			AConsoleDisplayPart[] cssArray;
  335 			if (ret.Length == 1)
  336 			{
  337 				cssArray = new AConsoleDisplayPart[cssList.Count];
  338 				cssList.CopyTo(cssArray);
  339 				if (bpList[0].CanSelect)
  340 					ret[0] = new ConsoleButtonString(parent, cssArray, bpList[0].Input);
  341 				else
  342 					ret[0] = new ConsoleButtonString(parent, cssArray);
  343 				return ret;
  344 			}
  345 			int cssStartCharIndex = 0;
  346 			int buttonEndCharIndex = 0;
  347 			int cssIndex = 0;
  348 			List<AConsoleDisplayPart> buttonCssList = new List<AConsoleDisplayPart>();
  349 			for (int i = 0; i < ret.Length; i++)
  350 			{
  351 				ButtonPrimitive bp = bpList[i];
  352 				buttonEndCharIndex += bp.Str.Length;
  353 				while (true)
  354 				{
  355 					if (cssIndex >= cssList.Count)
  356 						break;
  357 					AConsoleDisplayPart css = cssList[cssIndex];
  358 					if (cssStartCharIndex + css.Str.Length >= buttonEndCharIndex)
  359 					{//ボタンの終端を発見
  360 						int used = buttonEndCharIndex - cssStartCharIndex;
  361 						if (used > 0 && css.CanDivide)
  362 						{//cssの区切りの途中でボタンの区切りがある。
  363 							
  364 							ConsoleStyledString newCss = ((ConsoleStyledString)css).DivideAt(used);
  365 							if (newCss != null)
  366 							{
  367 								cssList.Insert(cssIndex + 1, newCss);
  368 								newCss.PointX = css.PointX + css.Width;
  369 							}
  370 						}
  371 						buttonCssList.Add(css);
  372 						cssStartCharIndex += css.Str.Length;
  373 						cssIndex++;
  374 						break;
  375 					}
  376 					//ボタンの終端はまだ先。
  377 					buttonCssList.Add(css);
  378 					cssStartCharIndex += css.Str.Length;
  379 					cssIndex++;
  380 				}
  381 				cssArray = new AConsoleDisplayPart[buttonCssList.Count];
  382 				buttonCssList.CopyTo(cssArray);
  383 				if (bp.CanSelect)
  384 					ret[i] = new ConsoleButtonString(parent, cssArray, bp.Input);
  385 				else
  386 					ret[i] = new ConsoleButtonString(parent, cssArray);
  387 				buttonCssList.Clear();
  388 			}
  389 			return ret;
