   33 				{//1753 InstructionLineを先に持ってきてみる。わずかに速くなった気がしないでもない
   34 					if (!Program.DebugMode && func.Function.IsDebug())
   35 					{//非DebugモードでのDebug系命令。何もしない。（SIF文のためにコメント行扱いにはできない）
   36 						continue;
   37 					}

   39 					{
   40 						ArgumentParser.SetArgumentTo(func);

   45 					{

   47 						{

   69 					//1750 飛んできた直後にShiftNextが入るのでここが実行されることは無いはず
   70 					//else//CALLやJUMPで飛んできた
   71 					//return;
   72 				}
   73 				else if (line is GotoLabelLine)
   74 					continue;//＄ラベル。何もすることはない。
   75 				else if (line is InvalidLine)
   76 				{
   77 					if (string.IsNullOrEmpty(line.ErrMes))
   78 						throw new CodeEE("読込に失敗した行が実行されました。エラーの詳細は読込時の警告を参照してください。");

   80 						throw new CodeEE(line.ErrMes);
   81 				}
   82 				//現在そんなものはない
   83 				//else
   84 				//	throw new ExeEE("定義されていない種類の行です");
   85 				if (!console.IsRunning || state.ScriptEnd)

  643 						FixedVariableTerm p = arrayArg.VarToken.GetFixedVariableTerm(exm);
  644 						int start = (int)arrayArg.Num1.GetIntValue(exm);

  649 						{
  650 							num = (int)arrayArg.Num2.GetIntValue(exm);
  651 							if (num < 0)

  662 					{
  663 						SpCopyArrayArgument arrayArg = (SpCopyArrayArgument)func.Argument;
  664 						IOperandTerm varName1 = arrayArg.VarName1;
  665 						IOperandTerm varName2 = arrayArg.VarName2;
  666 						VariableToken[] vars = new VariableToken[2] { null, null };
  667 						if (!(varName1 is SingleTerm) || !(varName2 is SingleTerm))
  668 						{
  669 							string[] names = new string[2] { null, null };
  670 							names[0] = varName1.GetStrValue(exm);
  671 							names[1] = varName2.GetStrValue(exm);
  672 							if ((vars[0] = GlobalStatic.IdentifierDictionary.GetVariableToken(names[0], null, true)) == null)
  673 								throw new CodeEE("ARRAYCOPY命令の第１引数(" + names[0] + ")が有効な変数名ではありません");
  674 							if (!vars[0].IsArray1D && !vars[0].IsArray2D && !vars[0].IsArray3D)
  675 								throw new CodeEE("ARRAYCOPY命令の第１引数\"" + names[0] + "\"は配列変数ではありません");
  676 							if (vars[0].IsCharacterData)
  677 								throw new CodeEE("ARRAYCOPY命令の第１引数\"" + names[0] + "\"はキャラクタ変数です（対応していません）");
  678 							if ((vars[1] = GlobalStatic.IdentifierDictionary.GetVariableToken(names[1], null, true)) == null)
  679 								throw new CodeEE("ARRAYCOPY命令の第２引数(" + names[0] + ")が有効な変数名ではありません");
  680 							if (!vars[1].IsArray1D && !vars[1].IsArray2D && !vars[1].IsArray3D)
  681 								throw new CodeEE("ARRAYCOPY命令の第２引数\"" + names[1] + "\"は配列変数ではありません");
  682 							if (vars[1].IsCharacterData)
  683 								throw new CodeEE("ARRAYCOPY命令の第２引数\"" + names[1] + "\"はキャラクタ変数です（対応していません）");
  684 							if (vars[1].IsConst)
  685 								throw new CodeEE("ARRAYCOPY命令の第２引数\"" + names[1] + "\"は値を変更できない変数です");
  686 							if ((vars[0].IsArray1D && !vars[1].IsArray1D) || (vars[0].IsArray2D && !vars[1].IsArray2D) || (vars[0].IsArray3D && !vars[1].IsArray3D))
  687 								throw new CodeEE("ARRAYCOPY命令の２つの配列変数の次元数が一致していません");
  688 							if ((vars[0].IsInteger && vars[1].IsString) || (vars[0].IsString && vars[1].IsInteger))
  689 								throw new CodeEE("ARRAYCOPY命令の２つの配列変数の型が一致していません");
  690 						}
  691 						else
  692 						{
  693 							vars[0] = GlobalStatic.IdentifierDictionary.GetVariableToken(((SingleTerm)varName1).Str, null, true);
  694 							vars[1] = GlobalStatic.IdentifierDictionary.GetVariableToken(((SingleTerm)varName2).Str, null, true);
  695 							if ((vars[0].IsInteger && vars[1].IsString) || (vars[0].IsString && vars[1].IsInteger))
  696 								throw new CodeEE("ARRAYCOPY命令の２つの配列変数の型が一致していません");
  697 						}
  698 						vEvaluator.CopyArray(vars[0], vars[1]);
  699 					}
  700 					break;
  701 				case FunctionCode.ENCODETOUNI:
  702 					{
  703 						//int length = Encoding.UTF32.GetEncoder().GetByteCount(target.ToCharArray(), 0, target.Length, false);
  704 						//byte[] bytes = new byte[length];
  705 						//Encoding.UTF32.GetEncoder().GetBytes(target.ToCharArray(), 0, target.Length, bytes, 0, false);
  706 						//vEvaluator.setEncodingResult(bytes);
  707 						term = ((ExpressionArgument)func.Argument).Term;
  708 						string target = term.GetStrValue(exm);
  709 
  710 						int length = vEvaluator.RESULT_ARRAY.Length;
  711 						// result:0には長さが入るのでその分-1
  712 						if (target.Length > length - 1)
  713 							throw new CodeEE(String.Format("ENCODETOUNIの引数が長すぎます（現在{0}文字。最大{1}文字まで）", target.Length, length - 1));
  714 
  715 						int[] ary = new int[target.Length];
  716 						for (int i = 0; i < target.Length; i++)
  717 							ary[i] = char.ConvertToUtf32(target, i);
  718 						vEvaluator.SetEncodingResult(ary);
  719 					}
  720 					break;
  721 				case FunctionCode.ASSERT:

  723 						throw new CodeEE("ASSERT文の引数が0です");
  724 					break;
  725 				case FunctionCode.THROW:
  726 					throw new CodeEE(((ExpressionArgument)func.Argument).Term.GetStrValue(exm));
  727 				case FunctionCode.CLEARTEXTBOX:
  728 					GlobalStatic.MainWindow.clear_richText();
  729 					break;
  730 				case FunctionCode.STRDATA:
  731 					{
  732 						//表示データが空なら何もしないで飛ぶ
  733 						if (func.dataList.Count == 0)
  734 						{
  735 							state.JumpTo(func.JumpTo);
  736 							return;
  737 						}
  738 						int count = func.dataList.Count;
  739 						int choice = (int)exm.VEvaluator.GetNextRand(count);
  740 						List<InstructionLine> iList = func.dataList[choice];
  741 						int i = 0;
  742 						foreach (InstructionLine selectedLine in iList)
  743 						{
  744 							state.CurrentLine = selectedLine;
  745 							if (selectedLine.Argument == null)

  747 							term = ((ExpressionArgument)selectedLine.Argument).Term;

  749 							if (++i < (int)iList.Count)
  750 								str += "\n";
  751 						}
  752 						((StrDataArgument)func.Argument).Var.SetValue(str, exm);
  753 						//ジャンプするが、流れが連続であることを保証。
  754 						state.JumpTo(func.JumpTo);
  755 						break;
  756 					}
  757 #if DEBUG
  758 				default:
  759 					throw new ExeEE("未定義の関数");
  760 #endif
  761 			}
  762 			return;
  763 		}

  772 		bool doFlowControlFunction(InstructionLine func)
  773 		{
  774 			switch (func.FunctionCode)
  775 			{
  776 				case FunctionCode.LOADDATA:
  777 					{
  778 						ExpressionArgument intExpArg = (ExpressionArgument)func.Argument;
  779 						Int64 target = intExpArg.Term.GetIntValue(exm);
  780 						if (target < 0)
  781 							throw new CodeEE("LOADDATAの引数に負の値(" + target.ToString() + ")が指定されました");
  782 						else if (target > int.MaxValue)
  783 							throw new CodeEE("LOADDATAの引数(" + target.ToString() + ")が大きすぎます");
  784 						//EraDataResult result = vEvaluator.checkData((int)target);
  785 						EraDataResult result = vEvaluator.CheckData((int)target, EraSaveFileType.Normal);
  786 						if (result.State != EraDataState.OK)
  787 							throw new CodeEE("不正なデータをロードしようとしました");
  788 
  789 						if (!vEvaluator.LoadFrom((int)target))
  790 							throw new ExeEE("ファイルのロード中に予期しないエラーが発生しました");
  791 						state.ClearFunctionList();
  792 						state.SystemState = SystemStateCode.LoadData_DataLoaded;
  793 						return false;
  794 					}
  795 
  796 				case FunctionCode.TRYCALLLIST:
  797 				case FunctionCode.TRYJUMPLIST:
  798 					{
  799 						//if (!sequential)//RETURNで帰ってきた
  800 						//{
  801 						//	state.JumpTo(func.JumpTo);
  802 						//	break;
  803 						//}
  804 						string funcName = "";
  805 						CalledFunction callto = null;
  806 						SpCallArgment cfa = null;
  807 						foreach (InstructionLine iLine in func.callList)
  808 						{
  809 
  810 							cfa = (SpCallArgment)iLine.Argument;
  811 							funcName = cfa.FuncnameTerm.GetStrValue(exm);
  812 							if (Config.ICFunction)
  813 								funcName = funcName.ToUpper();
  814 							callto = CalledFunction.CallFunction(this, funcName, func.JumpTo);
  815 							if (callto == null)
  816 								continue;
  817 							callto.IsJump = func.Function.IsJump();
  818 							string errMes;
  819 							UserDefinedFunctionArgument args = callto.ConvertArg(cfa.RowArgs, out errMes);
  820 							if (args == null)
  821 								throw new CodeEE(errMes);
  822 							state.IntoFunction(callto, args, exm);
  823 							return true;
  824 						}
  825 						state.JumpTo(func.JumpTo);
  826 					}
  827 					break;
  828 				case FunctionCode.TRYGOTOLIST:
  829 					{
  830 						string funcName = "";
  831 						LogicalLine jumpto = null;
  832 						foreach (InstructionLine iLine in func.callList)
  833 						{
  834 							if (iLine.Argument == null)
  835 								ArgumentParser.SetArgumentTo(iLine);
  836 							funcName = ((SpCallArgment)iLine.Argument).FuncnameTerm.GetStrValue(exm);
  837 							if (Config.ICVariable)
  838 								funcName = funcName.ToUpper();
  839 							jumpto = state.CurrentCalled.CallLabel(this, funcName);
  840 							if (jumpto != null)
  841 								break;
  842 						}
  843 						if (jumpto == null)
  844 							state.JumpTo(func.JumpTo);
  845 						else
  846 							state.JumpTo(jumpto);
  847 					}
  848 					break;
  849 				case FunctionCode.CALLTRAIN:
  850 					{
  851 						ExpressionArgument intExpArg = (ExpressionArgument)func.Argument;
  852 						Int64 count = intExpArg.Term.GetIntValue(exm);
  853 						SetCommnds(count);
  854 						return false;
  855 					}
  856 				case FunctionCode.STOPCALLTRAIN:
  857 					{
  858 						if (isCTrain)
  859 						{
  860 							ClearCommands();
  861 							skipPrint = false;
  862 						}
  863 						return false;
  864 					}
  865 				case FunctionCode.DOTRAIN:
  866 					{
  867 						switch (state.SystemState)
  868 						{
  869 							//case SystemStateCode.Train_Begin://BEGIN TRAINから。
  870 							case SystemStateCode.Train_CallEventTrain://@EVENTTRAINの呼び出し中。スキップ可能
  871 							case SystemStateCode.Train_CallShowStatus://@SHOW_STATUSの呼び出し中
  872 							//case SystemStateCode.Train_CallComAbleXX://@COM_ABLExxの呼び出し中。
  873 							case SystemStateCode.Train_CallShowUserCom://@SHOW_USERCOMの呼び出し中
  874 							//case SystemStateCode.Train_WaitInput://入力待ち状態。選択が実行可能ならEVENTCOMからCOMxx、そうでなければ@USERCOMにRESULTを渡す
  875 							//case SystemStateCode.Train_CallEventCom://@EVENTCOMの呼び出し中
  876 							//case SystemStateCode.Train_CallComXX://@COMxxの呼び出し中
  877 							//case SystemStateCode.Train_CallSourceCheck://@SOURCE_CHECKの呼び出し中
  878 							case SystemStateCode.Train_CallEventComEnd://@EVENTCOMENDの呼び出し中。スキップ可能。Train_CallEventTrainへ帰る。@USERCOMの呼び出し中もここ
  879 								break;
  880 							default:
  881 								exm.Console.PrintSystemLine(state.SystemState.ToString());
  882 								throw new CodeEE("DOTRAIN命令をこの位置で実行することはできません");
  883 						}
  884 						coms.Clear();
  885 						isCTrain = false;
  886 						this.count = 0;
  887 
  888 						Int64 train = ((ExpressionArgument)func.Argument).Term.GetIntValue(exm);
  889 						if (train < 0)
  890 							throw new CodeEE("DOTRAIN命令に0未満の値が渡されました");
  891 						if (train >= TrainName.Length)
  892 							throw new CodeEE("DOTRAIN命令にTRAINNAMEの配列数以上の値が渡されました");
  893 						doTrainSelectCom = train;
  894 						state.SystemState = SystemStateCode.Train_DoTrain;
  895 						return false;
  896 					}
  897 #if DEBUG
  898 				default:
  899 					throw new ExeEE("未定義の関数です");
  900 #endif
  901 			}
  902 			return true;
  903 		}

  914 			if (state != null)
  915 			{
  916 				prevStateList.Add(state);
  917 				state = state.Clone();
  918 			}
  919 		}

  921 		public void loadPrevState()
  922 		{
  923 			//怖いところだが、現状起こらない現象なので一旦消してみる
  924 			//if (prevStateList.Count == 0)

  926 			if (state != null)
  927 			{
  928 				state.ClearFunctionList();
  929 				state = prevStateList[prevStateList.Count - 1];
  930 				deletePrevState();
  931 			}
  932 		}

  934 		private void deletePrevState()
  935 		{
  936 			if (prevStateList.Count == 0)
  937 				return;
  938 			prevStateList.RemoveAt(prevStateList.Count - 1);
  939 		}

  941 		private void deleteAllPrevState()
  942 		{
  943 			foreach (ProcessState state in prevStateList)
  944 				state.ClearFunctionList();
  945 			prevStateList.Clear();
  946 		}
