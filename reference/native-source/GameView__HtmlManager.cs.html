  230 		{
  231 			List<string> strList = new List<string>();
  232 			StringStream st = new StringStream(str);
  233 			int found;
  234 			while (!st.EOS)
  235 			{
  236 				found = st.Find('<');
  237 				if (found < 0)
  238 				{
  239 					strList.Add(st.Substring());
  240 					break;
  241 				}
  242 				else if (found > 0)
  243 				{
  244 					strList.Add(st.Substring(st.CurrentPosition, found));
  245 					st.CurrentPosition += found;
  246 				}
  247 				found = st.Find('>');
  248 				if(found < 0)
  249 					return null;
  250 				found++;
  251 				strList.Add(st.Substring(st.CurrentPosition, found));
  252 				st.CurrentPosition += found;
  253 			}
  254 			string[] ret = new string[strList.Count];
  255 			strList.CopyTo(ret);
  256 			return ret;

  267 		{
  268 			List<AConsoleDisplayPart> cssList = new List<AConsoleDisplayPart>();

  270 			StringStream st = new StringStream(str);
  271 			int found;
  272 			bool hasComment = str.IndexOf("<!--") >= 0;
  273 			bool hasReturn = str.IndexOf('\n') >= 0;
  274 			HtmlAnalzeState state = new HtmlAnalzeState();
  275 			while (!st.EOS)
  276 			{
  277 				found = st.Find('<');
  278 				if (hasReturn)
  279 				{
  280 					int rFound = st.Find('\n');
  281 					if (rFound >= 0 && (found > rFound || found < 0))
  282 						found = rFound;
  283 				}
  284 				if (found < 0)
  285 				{
  286 					string txt = Unescape(st.Substring());
  287 					cssList.Add(new ConsoleStyledString(txt, state.GetSS()));
  288 					if (state.FlagPClosed)
  289 						throw new CodeEE("</p>の後にテキストがあります");
  290 					if (state.FlagNobrClosed)
  291 						throw new CodeEE("</nobr>の後にテキストがあります");
  292 					break;
  293 				}
  294 				else if (found > 0)
  295 				{
  296 					string txt = Unescape(st.Substring(st.CurrentPosition, found));
  297 					cssList.Add(new ConsoleStyledString(txt, state.GetSS()));
  298 					state.LineHead = false;
  299 					st.CurrentPosition += found;
  300 				}
  301 				//コメントタグのみ特別扱い
  302 				if (hasComment && st.CurrentEqualTo("<!--"))
  303 				{
  304 					st.CurrentPosition += 4;
  305 					found = st.Find("-->");
  306 					if (found < 0)
  307 						throw new CodeEE("コメンdト終了タグ\"-->\"がみつかりません");
  308 					st.CurrentPosition += found + 3;
  309 					continue;
  310 				}
  311 				if (hasReturn && st.Current == '\n')//テキスト中の\nは<br>として扱う
  312 				{
  313 					state.FlagBr = true;
  314 					st.ShiftNext();
  315 				}
  316 				else//タグ解析
  317 				{
  318 					st.ShiftNext();
  319 					AConsoleDisplayPart part = tagAnalyze(state, st);
  320 					if (st.Current != '>')
  321 						throw new CodeEE("タグ終端'>'が見つかりません");
  322 					if (part != null)
  323 						cssList.Add(part);
  324 					st.ShiftNext();
  325 				}
  326 
  327 				if (state.FlagBr)
  328 				{
  329 					state.LastButtonTag = state.CurrentButtonTag;
  330 					if (cssList.Count > 0)
  331 						buttonList.Add(cssToButton(cssList, state, console));
  332 					buttonList.Add(null);
  333 				}
  334 				if (state.FlagButton && cssList.Count > 0)
  335 				{
  336 					buttonList.Add(cssToButton(cssList, state, console));
  337 				}
  338 				state.FlagBr = false;
  339 				state.FlagButton = false;
  340 				state.LastButtonTag = state.CurrentButtonTag;
  341 			}
  342 			//</nobr></p>は省略許可
  343 			if (state.CurrentButtonTag != null || state.FontStyle != FontStyle.Regular || state.FonttagList.Count > 0)
  344 				throw new CodeEE("閉じられていないタグがあります");
  345 			if (cssList.Count > 0)
  346 				buttonList.Add(cssToButton(cssList, state, console));
  347 
  348 			foreach(ConsoleButtonString button in buttonList)
  349 			{
  350 				if (button != null && button.PointXisLocked)
  351 				{
  352 					if (!state.FlagNobr)
  353 						throw new CodeEE("<nobr>が設定されていない行ではpos属性は使用できません");
  354 					if (state.Alignment != DisplayLineAlignment.LEFT)
  355 						throw new CodeEE("alignがleftでない行ではpos属性は使用できません");
  356 					break;
  357 				}
  358 			}
  359 			ConsoleDisplayLine[] ret = PrintStringBuffer.ButtonsToDisplayLines(buttonList, sm, state.FlagNobr, false);
  360 
  361 			foreach (ConsoleDisplayLine dl in ret)
  362 			{
  363 				dl.SetAlignment(state.Alignment);
  364 			}
  365 			return ret;
  366 		}

  374 		public static string Escape(string str)
  375 		{
  376 			//Net4.5では便利なクラスがあるらしい
  377 			//return System.Web.HttpUtility.HtmlEncode(str);
  378 
  379 			int index = 0;
  380 			int found;
  381 			StringBuilder b = new StringBuilder();
  382 			while (index < str.Length)
  383 			{
  384 				found = str.IndexOfAny(rep, index);
  385 				if (found < 0)//見つからなければ以降を追加して終了
  386 				{
  387 					b.Append(str.Substring(index));
  388 					break;
  389 				}
  390 				if (found > index)//間に非エスケープ文字があるなら追加しておく
  391 					b.Append(str.Substring(index, found - index));
  392 				string repnew = repDic[str[found]];
  393 				b.Append(repnew);
  394 				index = found + 1;
  395 			}
  396 			return b.ToString();
  397 		}

  399 		public static string Unescape(string str)
  400 		{
  401 			int index = 0;
  402 			int found = str.IndexOf('&', index);
  403 			if (found < 0)
  404 				return str;
  405 			StringBuilder b = new StringBuilder();
  406 			// &～; をひたすら置換するだけ
  407 			while (index < str.Length)
  408 			{
  409 				found = str.IndexOf('&', index);
  410 				if (found < 0)//見つからなければ以降を追加して終了
  411 				{
  412 					b.Append(str.Substring(index));
  413 					break;
  414 				}
  415 				if (found > index)//間に非エスケープ文字があるなら追加しておく
  416 					b.Append(str.Substring(index, found - index));
  417 				index = found;
  418 				found = str.IndexOf(';', index);
  419 				if (found <= index + 1)
  420 				{
  421 					if (found < 0)
  422 						throw new CodeEE("'&'に対応する';'がみつかりません");
  423 					throw new CodeEE("'&'と';'が連続しています");
  424 				}
  425 				string escWordRow = str.Substring(index + 1, found - index - 1);
  426 				index = found + 1;
  427 				string escWord = escWordRow.ToLower();
  428 				int unicode;
  429 				switch (escWord)
  430 				{
  431 					case "nbsp": b.Append(" "); break;
  432 					case "amp": b.Append("&"); break;
  433 					case "gt": b.Append(">"); break;
  434 					case "lt": b.Append("<"); break;
  435 					case "quot": b.Append("\""); break;
  436 					case "apos": b.Append("\'"); break;
  437 					default:
  438 						{
  439 							int iBbase = 10;
  440 							if (escWord[0] != '#')
  441 								throw new CodeEE("\"&" + escWordRow + ";\"は適切な文字参照ではありません");
  442 							if (escWord.Length > 1 && escWord[1] == 'x')
  443 							{
  444 								iBbase = 16;
  445 								escWord = escWord.Substring(2);
  446 							}
  447 							else
  448 								escWord = escWord.Substring(1);
  449 							try
  450 							{
  451 								unicode = Convert.ToInt32(escWord, iBbase);
  452 							}
  453 							catch
  454 							{
  455 
  456 								throw new CodeEE("\"&" + escWordRow + ";\"は適切な文字参照ではありません");
  457 							}
  458 
  459 							if (unicode < 0 || unicode > 0xFFFF)
  460 								throw new CodeEE("\"&" + escWordRow + ";\"はUnicodeの範囲外です(サロゲートペアは使えません)");
  461 							b.Append((char)unicode);
  462 							break;
  463 						}
  464 				}
  465 			}
  466 			return b.ToString();
  467 		}

  988 		private static int stringToColorInt32(string str)
  989 		{
