   83 		public string CheckEscape(string str)
   84 		{

   86 			StringBuilder buffer = new StringBuilder();
   87 
   88 			while (!st.EOS)
   89 			{
   90 				//エスケープ文字の使用
   91 				if (st.Current == '\\')
   92 				{
   93 					st.ShiftNext();
   94 					switch (st.Current)
   95 					{
   96 						case '\\':
   97 							buffer.Append('\\');
   98 							buffer.Append('\\');
   99 							break;
  100 						case '{':
  101 						case '}':
  102 						case '%':
  103 						case '@':
  104 							buffer.Append('\\');
  105 							buffer.Append(st.Current);
  106 							break;
  107 						default:
  108 							buffer.Append("\\\\");
  109 							buffer.Append(st.Current);
  110 							break;
  111 					}
  112 					st.ShiftNext();
  113 					continue;
  114 				}
  115 				buffer.Append(st.Current);
  116 				st.ShiftNext();
  117 			}
  118 			return buffer.ToString();
  119 		}
