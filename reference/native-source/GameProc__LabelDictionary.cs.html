   58 		public void SortLabels()
   59 		{
   60 			foreach (KeyValuePair<string, List<FunctionLabelLine>[]> pair in eventLabelDic)
   61 				foreach (List<FunctionLabelLine> list in pair.Value)
   62 					list.Clear();

   64 			noneventLabelDic.Clear();

   66 			{
   67 				string key = pair.Key;
   68 				List<FunctionLabelLine> list = pair.Value;
   69 				if(list.Count > 1)
   70 					list.Sort();
   71 				if (!list[0].IsEvent)
   72 				{
   73 					noneventLabelDic.Add(key, list[0]);
   74                     GlobalStatic.IdentifierDictionary.resizeLocalVars("ARG", list[0].LabelName, list[0].ArgLength);
   75                     GlobalStatic.IdentifierDictionary.resizeLocalVars("ARGS", list[0].LabelName, list[0].ArgsLength);
   76 					continue;
   77 				}
   78 				//1810alpha010 オプションによりイベント関数をイベント関数でないかのように呼び出すことを許可
   79 				//eramaker仕様 - #PRI #LATER #SINGLE等を無視し、最先に定義された関数1つのみを呼び出す
   80 				if (Config.CompatiCallEvent)
   81 					noneventLabelDic.Add(key, list[0]);
   82 				List<FunctionLabelLine>[] eventLabels = new List<FunctionLabelLine>[4];
   83                 List<FunctionLabelLine> onlylist = new List<FunctionLabelLine>();
   84 				List<FunctionLabelLine> prilist = new List<FunctionLabelLine>();
   85 				List<FunctionLabelLine> normallist = new List<FunctionLabelLine>();
   86 				List<FunctionLabelLine> laterlist = new List<FunctionLabelLine>();
   87                 int localMax = 0;
   88                 int localsMax = 0;
   89 				for (int i = 0; i < list.Count; i++)
   90 				{
   91                     if (list[i].LocalLength > localMax)
   92                         localMax = list[i].LocalLength;
   93                     if (list[i].LocalsLength > localsMax)
   94                         localsMax = list[i].LocalsLength;
   95                     if (list[i].IsOnly)
   96                         onlylist.Add(list[i]);
   97 					if (list[i].IsPri)
   98 						prilist.Add(list[i]);
   99 					if (list[i].IsLater)
  100 						laterlist.Add(list[i]);//#PRIかつ#LATERなら二重に登録する。eramakerの仕様
  101 					if ((!list[i].IsPri) && (!list[i].IsLater))
  102 						normallist.Add(list[i]);
  103 				}
  104                 if (localMax < GlobalStatic.IdentifierDictionary.getLocalDefaultSize("LOCAL"))
  105                     localMax = GlobalStatic.IdentifierDictionary.getLocalDefaultSize("LOCAL");
  106                 if (localsMax < GlobalStatic.IdentifierDictionary.getLocalDefaultSize("LOCALS"))
  107                     localsMax = GlobalStatic.IdentifierDictionary.getLocalDefaultSize("LOCALS");
  108                 eventLabels[0] = onlylist;
  109 				eventLabels[1] = prilist;
  110 				eventLabels[2] = normallist;
  111 				eventLabels[3] = laterlist;
  112                 for (int i = 0; i < 4; i++)
  113                 {
  114                     for (int j = 0; j < eventLabels[i].Count; j++)
  115                     {
  116                         eventLabels[i][j].LocalLength = localMax;
  117                         eventLabels[i][j].LocalsLength = localsMax;
  118                     }
  119                 }
  120                 eventLabelDic.Add(key, eventLabels);
  121 			}

  144 		public void RemoveLabelWithPath(string fname)
  145 		{
  146 			List<FunctionLabelLine> labelLines;
  147 			List<FunctionLabelLine> removeLine = new List<FunctionLabelLine>();
  148 			List<string> removeKey = new List<string>();
  149 			foreach (KeyValuePair<string, List<FunctionLabelLine>> pair in labelAtDic)
  150 			{
  151 				string key = pair.Key;
  152 				labelLines = pair.Value;
  153 				foreach (FunctionLabelLine labelLine in labelLines)
  154 				{
  155 					if (string.Equals(labelLine.Position.Filename, fname, Config.SCIgnoreCase))
  156 						removeLine.Add(labelLine);
  157 				}
  158 				foreach (FunctionLabelLine remove in removeLine)
  159 				{
  160 					labelLines.Remove(remove);
  161 					if (labelLines.Count == 0)
  162 						removeKey.Add(key);
  163 				}
  164 				removeLine.Clear();
  165 			}
  166 			foreach (string rKey in removeKey)
  167 			{
  168 				labelAtDic.Remove(rKey);
  169 			}
  170 			for (int i = 0; i < invalidList.Count; i++)
  171 			{
  172 				if (string.Equals(invalidList[i].Position.Filename, fname, Config.SCIgnoreCase))
  173 				{
  174 					invalidList.RemoveAt(i);
  175 					i--;
  176 				}
  177 			}
  178 		}
