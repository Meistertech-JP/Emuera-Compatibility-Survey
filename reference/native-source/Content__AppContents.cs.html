   77 		static public bool LoadContents()
   78 		{
   79 			if (!Directory.Exists(Program.ContentDir))
   80 				return true;
   81 			try
   82 			{
   83 				//resourcesフォルダ内の全てのcsvファイルを探索する
   84 				string[] csvFiles = Directory.GetFiles(Program.ContentDir, "*.csv", SearchOption.AllDirectories);
   85 				foreach (var filepath in csvFiles)
   86 				{
   87 					//".csv"のみを拾うように
   88 					if (0 != ".csv".CompareTo(Path.GetExtension(filepath).ToLower()))
   89 						continue;
   90 					//アニメスプライト宣言。nullでないとき、フレーム追加モード
   91 					SpriteAnime currentAnime = null;
   92 					string directory = Path.GetDirectoryName(filepath).ToUpper() + "\\";
   93 					string filename = Path.GetFileName(filepath);
   94 					string[] lines = File.ReadAllLines(filepath, Config.Encode);
   95 					int lineNo = 0;
   96 					foreach (var line in lines)
   97 					{
   98 						lineNo++;
   99 						if (line.Length == 0)
  100 							continue;
  101 						string str = line.Trim();
  102 						if (str.Length == 0 || str.StartsWith(";"))
  103 							continue;
  104 						string[] tokens = str.Split(',');
  105 						//AContentItem item = CreateFromCsv(tokens);
  106 						ScriptPosition sp = new ScriptPosition(filename, lineNo);
  107                         if (CreateFromCsv(tokens, directory, currentAnime, sp) is ASprite item)
  108                         {
  109                             //アニメスプライト宣言ならcurrentAnime上書きしてフレーム追加モードにする。そうでないならnull
  110                             currentAnime = item as SpriteAnime;
  111                             if (!imageDictionary.ContainsKey(item.Name))
  112                             {
  113                                 imageDictionary.Add(item.Name, item);
  114                             }
  115                             else
  116                             {
  117                                 ParserMediator.Warn("同名のリソースがすでに作成されています:" + item.Name, sp, 0);
  118                                 item.Dispose();
  119                             }
  120                         }
  121                     }
  122 				}
  123 			}
  124 			catch
  125 			{
  126 				return false;
  127 				//throw new CodeEE("リソースファイルのロード中にエラーが発生しました");
  128 			}
  129 			return true;
  130 		}

  159 		static private AContentItem CreateFromCsv(string[] tokens, string dir, SpriteAnime currentAnime, ScriptPosition sp)
  160 		{
  161 			if(tokens.Length < 2)
  162 				return null;
  163 			string name = tokens[0].Trim().ToUpper();//
  164 			string arg2 = tokens[1].ToUpper();//画像ファイル名
  165 			if (name.Length == 0 || arg2.Length == 0)
  166 				return null;
  167 			//アニメーションスプライト宣言
  168 			if (arg2 == "ANIME")
  169 			{
  170 				if (tokens.Length < 4)
  171 				{
  172 					ParserMediator.Warn("アニメーションスプライトのサイズが宣言されていません", sp, 1);
  173 					return null;
  174 				}
  175 				//w,h
  176 				int[] sizeValue = new int[2];
  177 				bool sccs = true;
  178 				for (int i = 0; i < 2; i++)
  179 					sccs &= int.TryParse(tokens[i + 2], out sizeValue[i]);
  180 				if (!sccs || sizeValue[0] <= 0 || sizeValue[1] <= 0 || sizeValue[0] > AbstractImage.MAX_IMAGESIZE || sizeValue[1] > AbstractImage.MAX_IMAGESIZE)
  181 				{
  182 					ParserMediator.Warn("アニメーションスプライトのサイズの指定が適切ではありません", sp, 1);
  183 					return null;
  184 				}
  185 				SpriteAnime anime = new SpriteAnime(name, new Size(sizeValue[0],sizeValue[1]));
  186 
  187 				return anime;
  188 			}
  189 			//アニメ宣言以外（アニメ用フレーム含む
  190 
  191 			if(arg2.IndexOf('.') < 0)
  192 			{
  193 				ParserMediator.Warn("第二引数に拡張子がありません:" + arg2, sp, 1);
  194 				return null;
  195 			}
  196 			string parentName = dir + arg2;
  197 
  198 
  199 			//親画像のロードConstImage
  200 			if (!resourceDic.ContainsKey(parentName))
  201 			{
  202 				string filepath = parentName;
  203 				if (!File.Exists(filepath))
  204 				{
  205 					ParserMediator.Warn("指定された画像ファイルが見つかりませんでした:" + arg2, sp, 1);
  206 					return null;
  207 				}
  208 				Bitmap bmp = new Bitmap(filepath);
  209 				if (bmp == null)
  210 				{
  211 					ParserMediator.Warn("指定されたファイルの読み込みに失敗しました:" + arg2, sp, 1);
  212 					return null;
  213 				}
  214 				if (bmp.Width > AbstractImage.MAX_IMAGESIZE || bmp.Height > AbstractImage.MAX_IMAGESIZE)
  215 				{
  216 					//1824-2 すでに8192以上の幅を持つ画像を利用したバリアントが存在してしまっていたため、警告しつつ許容するように変更
  217 					//	bmp.Dispose();
  218 					ParserMediator.Warn("指定された画像ファイルの大きさが大きすぎます(幅及び高さを"+ AbstractImage.MAX_IMAGESIZE.ToString()+ "以下にすることを強く推奨します):" + arg2, sp, 1);
  219 					//return null;
  220 				}
  221 				ConstImage img = new ConstImage(parentName);
  222 				img.CreateFrom(bmp, Config.TextDrawingMode == TextDrawingMode.WINAPI);
  223 				if (!img.IsCreated)
  224 				{
  225 					ParserMediator.Warn("画像リソースの作成に失敗しました:" + arg2, sp, 1);
  226 					return null;
  227 				}
  228 				resourceDic.Add(parentName, img);
  229 			}
  230             if (!(resourceDic[parentName] is ConstImage parentImage) || !parentImage.IsCreated)
  231             {
  232                 ParserMediator.Warn("作成に失敗したリソースを元にスプライトを作成しようとしました:" + arg2, sp, 1);
  233                 return null;
  234             }
  235             Rectangle rect = new Rectangle(new Point(0, 0), parentImage.Bitmap.Size);
  236 			Point pos = new Point();
  237 			int delay = 1000;
  238 			//name,parentname, x,y,w,h ,offset_x,offset_y, delayTime
  239 			if(tokens.Length >= 6)//x,y,w,h
  240 			{
  241 				int[] rectValue = new int[4];
  242 				bool sccs = true;
  243 				for (int i = 0; i < 4; i++)
  244 					sccs &= int.TryParse(tokens[i + 2], out rectValue[i]);
  245 				if (sccs)
  246 				{
  247 					rect = new Rectangle(rectValue[0], rectValue[1], rectValue[2], rectValue[3]);
  248 					if (rect.Width <= 0 || rect.Height <= 0)
  249 					{
  250 						ParserMediator.Warn("スプライトの高さ又は幅には正の値のみ指定できます:" + name, sp, 1);
  251 						return null;
  252 					}
  253 					if (!rect.IntersectsWith(new Rectangle(0,0,parentImage.Bitmap.Width, parentImage.Bitmap.Height)))
  254 					{
  255 						ParserMediator.Warn("親画像の範囲外を参照しています:" + name, sp, 1);
  256 						return null;
  257 					}
  258 				}
  259 				if(tokens.Length >= 8)
  260 				{
  261 					sccs = true;
  262 					for (int i = 0; i < 2; i++)
  263 						sccs &= int.TryParse(tokens[i + 6], out rectValue[i]);
  264 					if (sccs)
  265 						pos = new Point(rectValue[0], rectValue[1]);
  266 					if (tokens.Length >= 9)
  267 					{
  268 						sccs = int.TryParse(tokens[8], out delay);
  269 						if (sccs && delay <= 0)
  270 						{
  271 							ParserMediator.Warn("フレーム表示時間には正の値のみ指定できます:" + name, sp, 1);
  272 							return null;
  273 						}
  274 					}
  275 				}
  276 			}
  277 			//既存のスプライトに対するフレーム追加
  278 			if (currentAnime != null && currentAnime.Name == name)
  279 			{
  280 				if(!currentAnime.AddFrame(parentImage, rect, pos, delay))
  281 				{
  282 					ParserMediator.Warn("アニメーションスプライトのフレームの追加に失敗しました:" + arg2, sp, 1);
  283 					return null;
  284 				}
  285 				return null;
  286 			}
  287 
  288 			//新規スプライト定義
  289 			ASprite image = new SpriteF(name, parentImage, rect, pos);
  290 			return image;
  291 		}
