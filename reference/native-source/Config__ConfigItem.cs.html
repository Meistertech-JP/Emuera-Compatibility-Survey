  251 		private bool tryStringToBool(string arg, ref bool p)
  252 		{
  253 			if (arg == null)
  254 				return false;
  255 			string str = arg.Trim();
  256 			if (Int32.TryParse(str, out int i))
  257 			{
  258 				p = (i != 0);
  259 				return true;
  260 			}
  261 			if (str.Equals("NO", StringComparison.CurrentCultureIgnoreCase)
  262 				|| str.Equals("FALSE", StringComparison.CurrentCultureIgnoreCase)
  263 				|| str.Equals("後", StringComparison.CurrentCultureIgnoreCase))//"単位の位置"用
  264 			{
  265 				p = false;
  266 				return true;
  267 			}
  268 			if (str.Equals("YES", StringComparison.CurrentCultureIgnoreCase)
  269 				|| str.Equals("TRUE", StringComparison.CurrentCultureIgnoreCase)
  270 				|| str.Equals("前", StringComparison.CurrentCultureIgnoreCase))
  271 			{
  272 				p = true;
  273 				return true;
  274 			}
  275 			throw new CodeEE("不正な指定です");
  276 		}
