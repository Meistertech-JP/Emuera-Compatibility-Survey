  279 		public SingleTerm GetConfigValueInERB(string text, ref string errMes)
  280 		{
  281 			AConfigItem item = ConfigData.Instance.GetItem(text);
  282 			if(item == null)
  283 			{
  284 				errMes = "文字列\"" + text + "\"は適切なコンフィグ名ではありません";

  286 			}

  288 			switch(item.Code)
  289 			{
  290 				//<bool>
  291 				case ConfigCode.AutoSave://"オートセーブを行なう"
  292 				case ConfigCode.MoneyFirst://"単位の位置"
  293 					if(item.GetValue<bool>())
  294 						term = new SingleTerm(1);
  295 					else
  296 						term = new SingleTerm(0);
  297 					break;
  298 				//<int>
  299 				case ConfigCode.WindowX:// "ウィンドウ幅"
  300 				case ConfigCode.PrintCPerLine:// "PRINTCを並べる数"
  301 				case ConfigCode.PrintCLength:// "PRINTCの文字数"
  302 				case ConfigCode.FontSize:// "フォントサイズ"
  303 				case ConfigCode.LineHeight:// "一行の高さ"
  304 				case ConfigCode.SaveDataNos:// "表示するセーブデータ数"
  305 				case ConfigCode.MaxShopItem:// "販売アイテム数"
  306 				case ConfigCode.ComAbleDefault:// "COM_ABLE初期値"
  307 					term = new SingleTerm(item.GetValue<int>());
  308 					break;
  309 				//<Color>
  310 				case ConfigCode.ForeColor://"文字色"
  311 				case ConfigCode.BackColor://"背景色"
  312 				case ConfigCode.FocusColor://"選択中文字色"
  313 				case ConfigCode.LogColor://"履歴文字色"
  314 					{
  315 						Color color = item.GetValue<Color>();
  316 						term = new SingleTerm( ((color.R * 256) + color.G) * 256 + color.B);
  317 					}
  318 					break;
  319 
  320 				//<Int64>
  321 				case ConfigCode.pbandDef:// "PBANDの初期値"
  322 				case ConfigCode.RelationDef:// "RELATIONの初期値"
  323 					term = new SingleTerm(item.GetValue<Int64>());
  324 					break;
  325 
  326 				//<string>
  327 				case ConfigCode.FontName:// "フォント名"
  328 				case ConfigCode.MoneyLabel:// "お金の単位"
  329 				case ConfigCode.LoadLabel:// "起動時簡略表示"
  330 				case ConfigCode.DrawLineString:// "DRAWLINE文字"
  331 				case ConfigCode.TitleMenuString0:// "システムメニュー0"
  332 				case ConfigCode.TitleMenuString1:// "システムメニュー1"
  333 				case ConfigCode.TimeupLabel:// "時間切れ表示"
  334 					term = new SingleTerm(item.GetValue<string>());
  335 					break;
  336 				
  337 				//<char>
  338 				case ConfigCode.BarChar1:// "BAR文字1"
  339 				case ConfigCode.BarChar2:// "BAR文字2"
  340 					term = new SingleTerm(item.GetValue<char>().ToString());

  351 			}
  352 			return term;

  414 		{
  415 			Config.ClearFont();

  441 		private bool loadConfig(string confPath, bool fix)
  442 		{
  443 			if (!File.Exists(confPath))
  444 				return false;
  445 			EraStreamReader eReader = new EraStreamReader(false);
  446 			if (!eReader.Open(confPath))
  447 				return false;
  448 			ScriptPosition pos = null;
  449 			try
  450 			{
  451 				string line = null;
  452 				//bool defineIgnoreWarningFiles = false;
  453 				while ((line = eReader.ReadLine()) != null)
  454 				{
  455 					if ((line.Length == 0) || (line[0] == ';'))
  456 						continue;
  457 					pos = new ScriptPosition(eReader.Filename, eReader.LineNo);
  458 					string[] tokens = line.Split(new char[] { ':' });
  459 					if (tokens.Length < 2)
  460 						continue;
  461 					AConfigItem item = GetConfigItem(tokens[0].Trim());
  462 					if (item != null)
  463 					{
  464 						//1806beta001 CompatiDRAWLINEの廃止、CompatiLinefeedAs1739へ移行
  465 						if(item.Code == ConfigCode.CompatiDRAWLINE)
  466 						{
  467 							item = GetConfigItem(ConfigCode.CompatiLinefeedAs1739);
  468 						}
  469 						//if ((item.Code == ConfigCode.IgnoreWarningFiles))
  470 						//{ 
  471 						//    if (!defineIgnoreWarningFiles)
  472 						//        (item.GetValue<List<string>>()).Clear();
  473 						//    defineIgnoreWarningFiles = true;
  474 						//    if ((item.Fixed) && (fix))
  475 						//        item.Fixed = false;
  476 						//}
  477 						
  478 						if (item.Code == ConfigCode.TextEditor)
  479 						{
  480 							//パスの関係上tokens[2]は使わないといけない
  481 							if (tokens.Length > 2)
  482 							{
  483 								if (tokens[2].StartsWith("\\"))
  484 									tokens[1] += ":" + tokens[2];
  485 								if (tokens.Length > 3)
  486 								{
  487 									for (int i = 3; i < tokens.Length; i++)
  488 									{
  489 										tokens[1] += ":" + tokens[i];
  490 									}
  491 								}
  492 							}
  493 						}
  494 						if (item.Code == ConfigCode.EditorArgument)
  495 						{
  496 							//半角スペースを要求する引数が必要なエディタがあるので別処理で
  497 							((ConfigItem<string>)item).Value = tokens[1];
  498 							continue;
  499 						}
  500                         if (item.Code == ConfigCode.MaxLog && Program.AnalysisMode)
  501                         {
  502                             //解析モード時はここを上書きして十分な長さを確保する
  503                             tokens[1] = "10000";
  504                         }
  505 						if ((item.TryParse(tokens[1])) && (fix))
  506 							item.Fixed = true;
  507 					}
  508 #if DEBUG
  509 					//else
  510 					//	throw new Exception("コンフィグファイルが変");
  511 #endif
  512 				}
  513 			}
  514 			catch (EmueraException ee)
  515 			{
  516 				ParserMediator.ConfigWarn(ee.Message, pos, 1, null);
  517 			}
  518 			catch (Exception exc)
  519 			{
  520 				ParserMediator.ConfigWarn(exc.GetType().ToString() + ":" + exc.Message, pos, 1, exc.StackTrace);
  521 			}
  522 			finally { eReader.Dispose(); }
  523 			return true;
  524 		}

  528 		public void LoadReplaceFile(string filename)
  529 		{
  530 			EraStreamReader eReader = new EraStreamReader(false);
  531 			if (!eReader.Open(filename))
  532 				return;
  533 			ScriptPosition pos = null;
  534 			try
  535 			{
  536 				string line = null;
  537 				while ((line = eReader.ReadLine()) != null)
  538 				{
  539 					if ((line.Length == 0) || (line[0] == ';'))
  540 						continue;
  541 					pos = new ScriptPosition(eReader.Filename, eReader.LineNo);
  542                     string[] tokens = line.Split(new char[] { ',', ':' });
  543 					if (tokens.Length < 2)
  544 						continue;
  545                     string itemName = tokens[0].Trim();
  546                     tokens[1] = line.Substring(tokens[0].Length + 1);
  547                     if (string.IsNullOrEmpty(tokens[1].Trim()))
  548                         continue;
  549                     AConfigItem item = GetReplaceItem(itemName);
  550                     if (item != null)
  551                         item.TryParse(tokens[1]);
  552 				}
  553 			}
  554 			catch (EmueraException ee)
  555 			{
  556 				ParserMediator.Warn(ee.Message, pos, 1);
  557 			}
  558 			catch (Exception exc)
  559 			{
  560 				ParserMediator.Warn(exc.GetType().ToString() + ":" + exc.Message, pos, 1, exc.StackTrace);
  561 			}
  562 			finally { eReader.Dispose(); }
  563 		}
