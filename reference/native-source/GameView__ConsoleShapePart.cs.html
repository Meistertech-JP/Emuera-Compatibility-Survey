   40 			int lineHeight = Config.FontSize;
   41 			float[] paramPixel = new float[param.Length];
   42 			for (int i = 0; i < param.Length; i++)
   43 			{
   44 				paramPixel[i] = ((float)param[i] * lineHeight) / 100f;
   45 			}
   46 			RectangleF rectF;
   47

   50 				case "space":

   52 					{

   54 						ret = new ConsoleSpacePart(rectF);

   62 					}
   63 					else if (paramPixel.Length == 4)
   64 					{

   66 						//1820a12 サイズ上限撤廃
   67 						if (rectF.X >= 0 && rectF.Width > 0 && rectF.Height > 0)
   68 						//	rectF.Y >= 0 && (rectF.Y + rectF.Height) <= lineHeight)
   69 						{
   70 							ret = new ConsoleRectangleShapePart(rectF);
   71 						}
   72 					}
   73 					break;
   74 				case "polygon":
   75 					break;
   76 			}
   77 			if (ret == null)
   78 			{
   79 				ret = new ConsoleErrorShapePart(sb.ToString());
   80 			}
   81 			ret.AltText = sb.ToString();
   82 			ret.Color = color;
   83 			ret.ButtonColor = bcolor;
   84 			ret.colorChanged = colorchanged;

   86 		}
